-- เมื่อกดปุ่ม V1
V1Button.MouseButton1Click:Connect(function()
    ScreenGui:Destroy() -- ปิด GUI ทันทีที่กดเลือก
    
    task.spawn(function()
        -- 1. ดึงซอร์สโค้ดธรรมดามาจาก GitHub
        local source = game:HttpGet("https://raw.githubusercontent.com/foreverspacexc-blip/Test/refs/heads/main/INVISBladeBallMain.txt")

        -- 2. แปลงโค้ดที่ดึงมาให้กลายเป็นเลขฐาน 16 (Hex) ทั้งหมด
        local hex = source:gsub(".", function(c)
            return string.format("%02X", string.byte(c))
        end)

        -- 3. สั่งรันโค้ดจากตัวแปร hex โดยตรง
        assert(loadstring(hex:gsub("%x%x", function(h)
            return string.char(tonumber(h, 16))
        end)))()
    end)
end)

-- เมื่อกดปุ่ม V2

end)
