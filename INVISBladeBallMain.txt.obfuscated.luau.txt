-- [[ CUSTOM COOL BLACK & RED LOADING GUI ]] --

local CoreGui = game:GetService("CoreGui")
local TweenService = game:GetService("TweenService")
local Players = game:GetService("Players")

-- สร้าง ScreenGui ป้องกันการลบเมื่อตาย
local LoadingScreen = Instance.new("ScreenGui")
LoadingScreen.Name = "InvisHubLoading"
LoadingScreen.Parent = CoreGui
LoadingScreen.IgnoreGuiInset = true

-- พื้นหลังหลัก (สีดำสนิท เท่ๆ)
local MainFrame = Instance.new("Frame")
MainFrame.Name = "MainFrame"
MainFrame.Size = UDim2.new(1, 0, 1, 0)
MainFrame.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
MainFrame.BorderSizePixel = 0
MainFrame.Parent = LoadingScreen

-- เอฟเฟกต์แสงไฟสีแดงด้านหลัง (Glow/Gradient)
local UIGradient = Instance.new("UIGradient")
UIGradient.Color = ColorSequence.new({
    ColorSequenceKeypoint.new(0, Color3.fromRGB(20, 20, 20)),
    ColorSequenceKeypoint.new(0.5, Color3.fromRGB(40, 10, 10)),
    ColorSequenceKeypoint.new(1, Color3.fromRGB(15, 15, 15))
})
UIGradient.Rotation = 45
UIGradient.Parent = MainFrame

-- วงกลมสีขาว (ไอคอนโหลดดิ้ง)
local LoadingCircle = Instance.new("Frame")
LoadingCircle.Name = "LoadingCircle"
LoadingCircle.AnchorPoint = Vector2.new(0.5, 0.5)
LoadingCircle.Position = UDim2.new(0.5, 0, 0.45, 0)
LoadingCircle.Size = UDim2.new(0, 80, 0, 80)
LoadingCircle.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
LoadingCircle.BorderSizePixel = 0
LoadingCircle.Parent = MainFrame

-- ทำให้เป็นวงกลมสมบูรณ์
local CircleCorner = Instance.new("UICorner")
CircleCorner.CornerRadius = UDim.new(1, 0)
CircleCorner.Parent = LoadingCircle

-- เส้นขอบเรืองแสงสีแดงรอบวงกลม
local UIStroke = Instance.new("UIStroke")
UIStroke.Color = Color3.fromRGB(255, 0, 0)
UIStroke.Thickness = 3
UIStroke.Transparency = 0
UIStroke.Parent = LoadingCircle

-- ข้อความไตเติ้ล
local TitleText = Instance.new("TextLabel")
TitleText.Size = UDim2.new(1, 0, 0, 40)
TitleText.Position = UDim2.new(0, 0, 0.55, 10)
TitleText.BackgroundTransparency = 1
TitleText.Text = "INVIS HUB"
TitleText.TextColor3 = Color3.fromRGB(255, 0, 0) -- สีแดงเด่นๆ
TitleText.TextSize = 28
TitleText.Font = Enum.Font.GothamBold
TitleText.TextSpacing = 6
TitleText.Parent = MainFrame

-- ข้อความสถานะ
local StatusText = Instance.new("TextLabel")
StatusText.Size = UDim2.new(1, 0, 0, 20)
StatusText.Position = UDim2.new(0, 0, 0.60, 15)
StatusText.BackgroundTransparency = 1
StatusText.Text = "Connecting to services..."
StatusText.TextColor3 = Color3.fromRGB(150, 150, 150)
StatusText.TextSize = 14
StatusText.Font = Enum.Font.Gotham
StatusText.Parent = MainFrame

-- [[ SYSTEM ANIMATIONS ]] --

-- อนิเมชั่นวงกลมหมุนและเต้นตามจังหวะ (Pulse & Rotate)
local rotateInfo = TweenInfo.new(1.5, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1)
local rotateTween = TweenService:Create(LoadingCircle, rotateInfo, {Rotation = 360})
rotateTween:Play()

local pulseInfo = TweenInfo.new(0.8, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true)
local pulseTween = TweenService:Create(LoadingCircle, pulseInfo, {Size = UDim2.new(0, 90, 0, 90)})
pulseTween:Play()

-- ฟังก์ชันจำลองการโหลดให้ดูสมจริงและเท่
local stages = {
    "Checking execution environment...",
    "Bypassing Anti-Cheat...",
    "Injecting InvisHub Core...",
    "Ready to Blade Ball!"
}

for i, status in ipairs(stages) do
    StatusText.Text = status
    task.wait(0.8) -- หน่วงเวลาแต่ละขั้นให้ดูเนียน
end

-- อนิเมชั่นตอนโหลดเสร็จ (Fade Out อย่างนุ่มนวล)
StatusText.Text = "Successfully Loaded!"
StatusText.TextColor3 = Color3.fromRGB(0, 255, 100) -- เปลี่ยนเป็นสีเขียวแป๊บนึงว่าผ่านแล้ว

task.wait(0.5)

local fadeInfo = TweenInfo.new(0.6, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local fadeMain = TweenService:Create(MainFrame, fadeInfo, {BackgroundTransparency = 1})
local fadeCircle = TweenService:Create(LoadingCircle, fadeInfo, {BackgroundTransparency = 1})
local fadeStroke = TweenService:Create(UIStroke, fadeInfo, {Transparency = 1})
local fadeTitle = TweenService:Create(TitleText, fadeInfo, {TextTransparency = 1})
local fadeStatus = TweenService:Create(StatusText, fadeInfo, {TextTransparency = 1})

fadeMain:Play()
fadeCircle:Play()
fadeStroke:Play()
fadeTitle:Play()
fadeStatus:Play()

fadeMain.Completed:Connect(function()
    LoadingScreen:Destroy() -- ลบ GUI ออกจากหน้าจอหลังจากจางหายไปแล้ว
end)

-- [[ RUN YOUR TARGET SCRIPT ]] --

pcall(function()
    loadstring(game:HttpGet("https://raw.githubusercontent.com/foreverspacexc-blip/Test/refs/heads/main/INVISBladeBallMain.txt"))()
end)
