-- ═══════════════════════════════════════════════════════════════
-- LOADING FUNCTION
-- ═══════════════════════════════════════════════════════════════

local function updateStatus(text, progress, duration)

    Status.Text = text

    local percent = math.floor(progress * 100)
    Percentage.Text = percent .. "%"

    tween(
        Bar,
        TweenInfo.new(
            duration,
            Enum.EasingStyle.Quad,
            Enum.EasingDirection.Out
        ),
        {
            Size = UDim2.new(progress, 0, 1, 0)
        }
    ):Play()

    task.wait(duration)
end
