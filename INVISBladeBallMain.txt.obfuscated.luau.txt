local UserInputService = game:GetService("UserInputService")
local CoreGui = game:GetService("CoreGui")

local LocalPlayer = Players.LocalPlayer

-- 🌈 THEME COLORS & PALETTE (Premium Cyber Neo Dark)
local THEME = {
    Background  = Color3.fromRGB(12, 12, 16),      -- ดำมืดลึกมิติ
    TopBar      = Color3.fromRGB(18, 18, 24),      -- แถบหัวข้อด้านบน
    ButtonBg    = Color3.fromRGB(24, 25, 32),      -- พื้นหลังปุ่มมาตรฐาน
    TextMain    = Color3.fromRGB(255, 255, 255),  -- ตัวหนังสือสีขาวใส
    InvisColor  = Color3.fromRGB(0, 240, 255),    -- สีฟ้านีออน (Invis Hub)
    StellarColor = Color3.fromRGB(160, 100, 255),  -- สีม่วงนีออน (Stellar Hub)
    BorderColor = Color3.fromRGB(35, 36, 46)       -- ขอบสเตติกปกติ
}

-- 🖼️ SCREEN GUI SETUP

ScreenGui.Name = "Invis_Stellar_Selector_" .. math.random(100, 999)
ScreenGui.ResetOnSpawn = false
ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

pcall(function()
    if gethui then ScreenGui.Parent = gethui()
    elseif CoreGui then ScreenGui.Parent = CoreGui
    else ScreenGui.Parent = LocalPlayer:WaitForChild("PlayerGui", 10) end
end)

-- 🖼️ MAIN FRAME WINDOW (CanvasGroup เพื่อทำอนิเมชันเปิดแบบเฟดลื่นๆ)
local MainFrame = Instance.new("CanvasGroup", ScreenGui)
MainFrame.Name = "MainFrame"
MainFrame.Size = UDim2.new(0, 320, 0, 220)
MainFrame.Position = UDim2.new(0.5, -160, 0.5, -110) -- กึ่งกลางหน้าจอพอดี
MainFrame.BackgroundColor3 = THEME.Background
MainFrame.GroupTransparency = 1 -- เริ่มจากโปร่งแสงเพื่อเปิดตัวด้วยอนิเมชัน
MainFrame.Visible = true
Instance.new("UICorner", MainFrame).CornerRadius = UDim.new(0, 16)

-- 🌈 เส้นขอบไฟวิ่ง RGB ไหลเชื่องช้าเพิ่มความพรีเมียมรอบกรอบหน้าต่างหลัก
local MainStroke = Instance.new("UIStroke", MainFrame)
MainStroke.Thickness = 2
MainStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
task.spawn(function()
    while MainStroke and MainStroke.Parent do
        for i = 0, 1, 0.004 do
            if MainStroke and MainStroke.Parent then
                MainStroke.Color = Color3.fromHSV(i, 0.7, 0.9)
            end
            task.wait(0.02)
        end
    end
end)

-- 📌 ระบบลากย้ายตำแหน่งหน้าต่าง (Smooth Drag System)
local mDrag, mStart, mPos
MainFrame.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        mDrag = true; mStart = i.Position; mPos = MainFrame.Position
    end
end)
UserInputService.InputChanged:Connect(function(i)
    if mDrag and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local d = i.Position - mStart
        MainFrame.Position = UDim2.new(mPos.X.Scale, mPos.X.Offset + d.X, mPos.Y.Scale, mPos.Y.Offset + d.Y)
    end
end)
UserInputService.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then mDrag = false end
end)

-- 📦 BUTTON HUB BUILDER FUNCTION (ฟังก์ชันสร้างปุ่มดีไซน์ไฮเอนด์)
local function createHubButton(name, positionY, accentColor, descriptionText)
    local btnContainer = Instance.new("Frame", MainFrame)
    btnContainer.Size = UDim2.new(1, -32, 0, 56)
    btnContainer.Position = UDim2.new(0, 16, 0, positionY)
    btnContainer.BackgroundColor3 = THEME.ButtonBg
    Instance.new("UICorner", btnContainer).CornerRadius = UDim.new(0, 10)
    
    local btnStroke = Instance.new("UIStroke", btnContainer)
    btnStroke.Thickness = 1.5
    btnStroke.Color = THEME.BorderColor
    btnStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border

    local mainBtn = Instance.new("TextButton", btnContainer)
    mainBtn.Size = UDim2.new(1, 0, 1, 0)
    mainBtn.BackgroundTransparency = 1
    mainBtn.Text = ""

    local titleLbl = Instance.new("TextLabel", btnContainer)
    titleLbl.Size = UDim2.new(1, -20, 0, 22)
    titleLbl.Position = UDim2.new(0, 12, 0, 8)
    titleLbl.BackgroundTransparency = 1
    titleLbl.Text = name
    titleLbl.TextColor3 = THEME.TextMain
    titleLbl.FontFace = Font.new("rbxasset://fonts/families/Montserrat.json", Enum.FontWeight.Bold)
    titleLbl.TextSize = 14
    titleLbl.TextXAlignment = Enum.TextXAlignment.Left

    local descLbl = Instance.new("TextLabel", btnContainer)
    descLbl.Size = UDim2.new(1, -20, 0, 16)
    descLbl.Position = UDim2.new(0, 12, 0, 28)
    descLbl.BackgroundTransparency = 1
    descLbl.Text = descriptionText
    descLbl.TextColor3 = Color3.fromRGB(130, 132, 145)
    descLbl.FontFace = Font.new("rbxasset://fonts/families/Montserrat.json", Enum.FontWeight.SemiBold)
    descLbl.TextSize = 10
    descLbl.TextXAlignment = Enum.TextXAlignment.Left

    local arrowIcon = Instance.new("ImageLabel", btnContainer)
    arrowIcon.AnchorPoint = Vector2.new(0.5, 0.5)
    arrowIcon.Position = UDim2.new(0.92, 0, 0.5, 0)
    arrowIcon.Size = UDim2.new(0, 14, 0, 14)
    arrowIcon.BackgroundTransparency = 1
    arrowIcon.Image = "rbxassetid://3944703587" -- ไอคอนลูกศรชี้ไปข้างหน้าสไตล์ Rise UI
    arrowIcon.ImageColor3 = Color3.fromRGB(150, 152, 165)

    -- อนิเมชัน Hover (เมาส์ชี้สลับสีเรืองแสงพรีเมียม)
    mainBtn.MouseEnter:Connect(function()
        TweenService:Create(btnStroke, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {Color = accentColor}):Play()
        TweenService:Create(titleLbl, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {TextColor3 = accentColor}):Play()
        TweenService:Create(arrowIcon, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {ImageColor3 = accentColor, Position = UDim2.new(0.93, 0, 0.5, 0)}):Play()
    end)
    mainBtn.MouseLeave:Connect(function()
        TweenService:Create(btnStroke, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {Color = THEME.BorderColor}):Play()
        TweenService:Create(titleLbl, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {TextColor3 = THEME.TextMain}):Play()
        TweenService:Create(arrowIcon, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {ImageColor3 = Color3.fromRGB(150, 152, 165), Position = UDim2.new(0.92, 0, 0.5, 0)}):Play()
    end)

    -- อนิเมชันกดจิ้มปุ่ม (Click Scale Elastic Effect)
    mainBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            TweenService:Create(btnContainer, TweenInfo.new(0.1, Enum.EasingStyle.Quad), {Size = UDim2.new(1, -40, 0, 52), Position = UDim2.new(0, 20, 0, positionY + 2)}):Play()
        end
    end)
    mainBtn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            TweenService:Create(btnContainer, TweenInfo.new(0.25, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out), {Size = UDim2.new(1, -32, 0, 56), Position = UDim2.new(0, 16, 0, positionY)}):Play()
        end
    end)

    return mainBtn
end

local InvisButton = createHubButton("⚡ INVIS HUB", 62, THEME.InvisColor, "fast execute and good auto parry")

-- อนิมชันเฟดปิดและทำลายหน้าต่างเมื่อกดเลือกสำเร็จ
local function closeAndDestroyMenu()
    local t = TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {GroupTransparency = 1, Size = UDim2.new(0, 280, 0, 180), Position = MainFrame.Position + UDim2.new(0, 20, 0, 20)})
    t:Play()
    t.Completed:Once(function()
        ScreenGui:Destroy()
    end)
end

-- เมื่อเลือกกดยิง [ INVIS HUB ]
InvisButton.MouseButton1Click:Connect(function()
    closeAndDestroyMenu()
    task.spawn(function()
        -- ดึงโค้ดเวอร์ชันดักจับ Bypass ล่าสุดจากคลังที่คุณเปลี่ยนพิกัด
        local source = game:HttpGet("https://raw.githubusercontent.com/foreverspacexc-blip/Test/refs/heads/main/INVISOnly.txt")
        
        -- ทำงานถอดรหัสรันระบบฐาน 16 เสถียรสูงตามโครงสร้างเดิมของคุณ
        local hex = source:gsub(".", function(c)
            return string.format("%02X", string.byte(c))
        end)
        
        assert(loadstring(hex:gsub("%x%x", function(h)
            return string.char(tonumber(h, 16))
        end)))()
    end)
end)

-- เมื่อเลือกกดยิง [ STELLAR HUB ]
StellarButton.MouseButton1Click:Connect(function()

    end)
end)
