mDrag, mStart, mPos = nil, nil, nil
MainFrame.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        mDrag = true; mStart = i.Position; mPos = MainFrame.Position
    end
end)
UserInputService.InputChanged:Connect(function(i)
    if mDrag and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        d = i.Position - mStart
        MainFrame.Position = UDim2.new(mPos.X.Scale, mPos.X.Offset + d.X, mPos.Y.Scale, mPos.Y.Offset + d.Y)
    end
end)
UserInputService.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then mDrag = false end
end)

function createHubButton(name, accentColor, descriptionText)
    local btnContainer = Instance.new("Frame", ScrollContainer)
    btnContainer.Size = UDim2.new(1, -32, 0, 56)
    btnContainer.BackgroundColor3 = THEME.ButtonBg
    Instance.new("UICorner", btnContainer).CornerRadius = UDim.new(0, 10)
    
    local btnStroke = Instance.new("UIStroke", btnContainer)
    btnStroke.Thickness = 1.5
    btnStroke.Color = THEME.BorderColor
    btnStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border

    local mainBtn = Instance.new("TextButton", btnContainer)
    mainBtn.Size = UDim2.new(1, 0, 1, 0)
    mainBtn.BackgroundTransparency = 1
    mainBtn.Text = ""

    local titleLbl = Instance.new("TextLabel", btnContainer)
    titleLbl.Size = UDim2.new(1, -20, 0, 22)
    titleLbl.Position = UDim2.new(0, 12, 0, 8)
    titleLbl.BackgroundTransparency = 1
    titleLbl.Text = name
    titleLbl.TextColor3 = THEME.TextMain
    titleLbl.FontFace = Font.new("rbxasset://fonts/families/Montserrat.json", Enum.FontWeight.Bold)
    titleLbl.TextSize = 14
    titleLbl.TextXAlignment = Enum.TextXAlignment.Left

    local descLbl = Instance.new("TextLabel", btnContainer)
    descLbl.Size = UDim2.new(1, -20, 0, 16)
    descLbl.Position = UDim2.new(0, 12, 0, 28)
    descLbl.BackgroundTransparency = 1
    descLbl.Text = descriptionText
    descLbl.TextColor3 = Color3.fromRGB(130, 132, 145)
    descLbl.FontFace = Font.new("rbxasset://fonts/families/Montserrat.json", Enum.FontWeight.SemiBold)
    descLbl.TextSize = 10
    descLbl.TextXAlignment = Enum.TextXAlignment.Left

    local arrowIcon = Instance.new("ImageLabel", btnContainer)
    arrowIcon.AnchorPoint = Vector2.new(0.5, 0.5)
    arrowIcon.Position = UDim2.new(0.92, 0, 0.5, 0)
    arrowIcon.Size = UDim2.new(0, 14, 0, 14)
    arrowIcon.BackgroundTransparency = 1
    arrowIcon.Image = "rbxassetid://3944703587"
    arrowIcon.ImageColor3 = Color3.fromRGB(150, 152, 165)

    mainBtn.MouseEnter:Connect(function()
        TweenService:Create(btnStroke, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {Color = accentColor}):Play()
        TweenService:Create(titleLbl, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {TextColor3 = accentColor}):Play()
        TweenService:Create(arrowIcon, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {ImageColor3 = accentColor, Position = UDim2.new(0.93, 0, 0.5, 0)}):Play()
    end)
    mainBtn.MouseLeave:Connect(function()
        TweenService:Create(btnStroke, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {Color = THEME.BorderColor}):Play()
        TweenService:Create(titleLbl, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {TextColor3 = THEME.TextMain}):Play()
        TweenService:Create(arrowIcon, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {ImageColor3 = Color3.fromRGB(150, 152, 165), Position = UDim2.new(0.92, 0, 0.5, 0)}):Play()
    end)

    mainBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            TweenService:Create(btnContainer, TweenInfo.new(0.1, Enum.EasingStyle.Quad), {Size = UDim2.new(1, -40, 0, 52)}):Play()
        end
    end)
    mainBtn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            TweenService:Create(btnContainer, TweenInfo.new(0.25, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out), {Size = UDim2.new(1, -32, 0, 56)}):Play()
        end
    end)

    return mainBtn
end

function closeAndDestroyMenu()
    t = TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {GroupTransparency = 1, Size = UDim2.new(0, 280, 0, 180), Position = MainFrame.Position + UDim2.new(0, 20, 0, 20)})
    t:Play()
    t.Completed:Once(function()
        ScreenGui:Destroy()
    end)
end
