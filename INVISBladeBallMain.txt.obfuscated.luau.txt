-- [[ INVIS HUB LOADING SCREEN ]] --

-- ดึง CoreGui ผ่าน gethui() เพื่อความปลอดภัย
local TargetGui = (gethui and gethui()) or game:GetService("CoreGui")

-- เคลียร์ GUI เก่าออกถ้ามีการรันซ้ำ
if TargetGui:FindFirstChild("InvisHubLoading") then
    TargetGui.InvisHubLoading:Destroy()
end

-- สร้าง ScreenGui
local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "InvisHubLoading"
ScreenGui.Parent = TargetGui
ScreenGui.IgnoreGuiInset = true -- คลุมเต็มจอถึงขอบบน
ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

-- สร้างหน้าจอสีดำ
local MainFrame = Instance.new("Frame")
MainFrame.Name = "MainFrame"
MainFrame.Parent = ScreenGui
MainFrame.BackgroundColor3 = Color3.fromRGB(5, 5, 5) -- สีดำสนิท
MainFrame.BorderSizePixel = 0
MainFrame.Size = UDim2.new(1, 0, 1, 0)
MainFrame.Position = UDim2.new(0, 0, 0, 0)
MainFrame.BackgroundTransparency = 1 -- ตั้งให้โปร่งใสก่อนเพื่อทำ Fade in

-- ข้อความ INVIS HUB
local Title = Instance.new("TextLabel")
Title.Parent = MainFrame
Title.BackgroundTransparency = 1
Title.Position = UDim2.new(0.5, -150, 0.5, -60)
Title.Size = UDim2.new(0, 300, 0, 50)
Title.Font = Enum.Font.GothamBold
Title.Text = "INVIS HUB"
Title.TextColor3 = Color3.fromRGB(255, 255, 255)
Title.TextSize = 45
Title.TextTransparency = 1

-- เอฟเฟกต์เรืองแสงเท่ๆ สีม่วงดาร์กๆ
local Glow = Instance.new("UIStroke")
Glow.Parent = Title
Glow.Color = Color3.fromRGB(120, 0, 255) 
Glow.Thickness = 2
Glow.Transparency = 1

-- ข้อความสถานะการโหลด
local StatusText = Instance.new("TextLabel")
StatusText.Parent = MainFrame
StatusText.BackgroundTransparency = 1
StatusText.Position = UDim2.new(0.5, -100, 0.5, 15)
StatusText.Size = UDim2.new(0, 200, 0, 20)
StatusText.Font = Enum.Font.Gotham
StatusText.Text = "Initializing..."
StatusText.TextColor3 = Color3.fromRGB(150, 150, 150)
StatusText.TextSize = 14
StatusText.TextTransparency = 1

-- กรอบหลอดโหลด
local LoadingBarBack = Instance.new("Frame")
LoadingBarBack.Parent = MainFrame
LoadingBarBack.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
LoadingBarBack.BorderSizePixel = 0
LoadingBarBack.Position = UDim2.new(0.5, -100, 0.5, 45)
LoadingBarBack.Size = UDim2.new(0, 200, 0, 4)
LoadingBarBack.BackgroundTransparency = 1

local UICorner = Instance.new("UICorner")
UICorner.CornerRadius = UDim.new(1, 0)
UICorner.Parent = LoadingBarBack

-- หลอดโหลด (สีขาว)
local LoadingBar = Instance.new("Frame")
LoadingBar.Parent = LoadingBarBack
LoadingBar.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
LoadingBar.BorderSizePixel = 0
LoadingBar.Size = UDim2.new(0, 0, 1, 0)
LoadingBar.BackgroundTransparency = 1

local UICorner2 = Instance.new("UICorner")
UICorner2.CornerRadius = UDim.new(1, 0)
UICorner2.Parent = LoadingBar

-- ระบบ Tween Animation (เฟดเข้า-ออก แบบนุ่มนวล)
local TweenService = game:GetService("TweenService")
local fadeInfo = TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

-- เริ่มเล่น Animation เฟดเข้า
TweenService:Create(MainFrame, fadeInfo, {BackgroundTransparency = 0}):Play()
task.wait(0.5)
TweenService:Create(Title, fadeInfo, {TextTransparency = 0}):Play()
TweenService:Create(Glow, fadeInfo, {Transparency = 0.3}):Play()
TweenService:Create(StatusText, fadeInfo, {TextTransparency = 0}):Play()
TweenService:Create(LoadingBarBack, fadeInfo, {BackgroundTransparency = 0}):Play()
TweenService:Create(LoadingBar, fadeInfo, {BackgroundTransparency = 0}):Play()
task.wait(0.5)

-- ฟังก์ชันรันหลอดโหลด
local function updateStatus(text, progress, delayTime)
    StatusText.Text = text
    TweenService:Create(LoadingBar, TweenInfo.new(delayTime, Enum.EasingStyle.Linear), {Size = UDim2.new(progress, 0, 1, 0)}):Play()
    task.wait(delayTime)
end

-- ลำดับการโหลดสถานะต่างๆ
updateStatus("Checking Security...", 0.3, 0.5)
updateStatus("Bypassing Anti-Cheat...", 0.6, 0.6)
updateStatus("Fetching Scripts...", 0.9, 0.7)

-- [[ รันสคริปต์หลักที่กำหนดไว้ ]] --
updateStatus("Executing...", 1, 0.4)

pcall(function()
    loadstring(game:HttpGet("https://raw.githubusercontent.com/foreverspacexc-blip/Test/refs/heads/main/INVISBladeBallMain.txt"))()
end)

-- แจ้งว่าโหลดเสร็จแล้ว
StatusText.Text = "Welcome to INVIS HUB"
StatusText.TextColor3 = Color3.fromRGB(255, 255, 255)
task.wait(1)

-- เริ่มเล่น Animation เฟดออก
TweenService:Create(Title, fadeInfo, {TextTransparency = 1}):Play()
TweenService:Create(Glow, fadeInfo, {Transparency = 1}):Play()
TweenService:Create(StatusText, fadeInfo, {TextTransparency = 1}):Play()
TweenService:Create(LoadingBarBack, fadeInfo, {BackgroundTransparency = 1}):Play()
TweenService:Create(LoadingBar, fadeInfo, {BackgroundTransparency = 1}):Play()
task.wait(0.5)

TweenService:Create(MainFrame, fadeInfo, {BackgroundTransparency = 1}):Play()
task.wait(0.5)

-- ลบ UI ทิ้งเพื่อไม่ให้กินทรัพยากรเครื่อง
ScreenGui:Destroy()
