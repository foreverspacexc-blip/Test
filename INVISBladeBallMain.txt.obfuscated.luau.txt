local function updateStatus(text, progress, duration)
    Status.Text = text
    Percentage.Text = math.floor(progress * 100) .. "%"
    tween(Bar, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size = UDim2.new(progress, 0, 1, 0)}):Play()
    task.wait(duration)
end

local TargetGui = (gethui and gethui()) or game:GetService("CoreGui")
local LoadingGui = TargetGui:FindFirstChild("InvisHubLoading")

if LoadingGui and LoadingGui:FindFirstChild("Background") then
    local TweenService = game:GetService("TweenService")
    local Background = LoadingGui.Background
    local Main = Background:FindFirstChild("Main")
    
    local Status = Main and Main:FindFirstChild("Status")
    local Percentage = Main and Main:FindFirstChild("Percentage")
    local Logo = Main and Main:FindFirstChild("Logo")
    local Hub = Main and Main:FindFirstChild("Hub")
    local BarBack = Main and Main:FindFirstChild("BarBack")
    local Bar = BarBack and BarBack:FindFirstChild("Bar")
    local Accent = Main and Main:FindFirstChild("Accent")
    local Border = Main and Main:FindFirstChild("UIStroke")
    local Glow = Background:FindFirstChild("Glow")

    -- อัปเดตสถานะเป็นเสร็จสิ้น
    if Status then 
        Status.Text = "Welcome to INVIS HUB" 
        Status.TextColor3 = Color3.fromRGB(255, 255, 255)
    end
    if Percentage then Percentage.Text = "100%" end
    
    if BarBack and Bar then
        TweenService:Create(Bar, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Size = UDim2.new(1, 0, 1, 0)
        }):Play()
    end

    task.wait(0.9)

    -- เอฟเฟกต์ Fade Out ข้อมูล
    local FadeOut = TweenInfo.new(0.5, Enum.EasingStyle.Circular, Enum.EasingDirection.InOut)
    if Logo then TweenService:Create(Logo, FadeOut, {TextTransparency = 1}):Play() end
    if Hub then TweenService:Create(Hub, FadeOut, {TextTransparency = 1}):Play() end
    if Status then TweenService:Create(Status, FadeOut, {TextTransparency = 1}):Play() end
    if Percentage then TweenService:Create(Percentage, FadeOut, {TextTransparency = 1}):Play() end
    if BarBack then TweenService:Create(BarBack, FadeOut, {BackgroundTransparency = 1}):Play() end
    if Bar then TweenService:Create(Bar, FadeOut, {BackgroundTransparency = 1}):Play() end
    if Accent then TweenService:Create(Accent, FadeOut, {BackgroundTransparency = 1}):Play() end
    if Border then TweenService:Create(Border, FadeOut, {Transparency = 1}):Play() end
    if Glow then TweenService:Create(Glow, FadeOut, {ImageTransparency = 1}):Play() end
