local function LinearInterp(a, b, t) return a + (b - a) * t end

local function GetCurveData(ball)
    if Is_Curved_Data[ball] then return Is_Curved_Data[ball] end
    local data = {
        Curving = 0, Lerp_Radians = 0, Last_Warping = 0,
        Dot = 0, Flips = 0, Flip_Time = 0,
        Backward_Dot = 0, Backward_Flips = 0, Backward_Flip_Time = 0,
        Returned = false, Target_Count = 0,
    }
    Is_Curved_Data[ball] = data

    local function updateFrom()
        local from = ball:GetAttribute("from")
        if from == LP.Name and data.First_Hit then
            data.Returned = true
        elseif from and from ~= "" and from ~= LP.Name then
            data.First_Hit = data.First_Hit or from
            data.Returned = false
        end
    end

    data.FromConn   = ball:GetAttributeChangedSignal("from"):Connect(updateFrom)
    data.TargetConn = ball:GetAttributeChangedSignal("target"):Connect(function()
        local currentTarget = ball:GetAttribute("target")
        if currentTarget ~= LP.Name then
            parryLocked[ball] = false
            lockTime[ball]    = 0
        end
        data.Target_Count = (data.Target_Count or 0) + 1
    end)
    updateFrom()
    return data
end

        distance = offset.Magnitude
        if speed < 0.001 or distance < 0.001 then return false end
        if not shared._TargetStartedDist[ball] then
            shared._TargetStartedDist[ball] = distance / 3
        end
        if distance <= shared._TargetStartedDist[ball] then return false end
        if hrp.Position.Y > ball_pos.Y + 1 then return false end

        dir_to_me = offset.Unit

        cd_data = GetCurveData(ball)

        dot_diff = dir_diff.Magnitude > 0.001 and math.clamp(dot - dir_to_me:Dot(dir_diff.Unit), -1, 1) or dot
        dot_thresh = 0.5 - ping_val / 1000
        angle_t = 20 * math.max(dot, 0)
        reach_time = math.max(0, distance / speed - ping_val / 1000)
        dist_thresh = 15 - math.min(distance / 1000, 15) + angle_t + speed_t
        radians = math.rad(math.asin(math.clamp(dot, -1, 1)))
        time_now = os.clock()
        if dot == nil then return false end
        if math.sign(dot) ~= math.sign(cd_data.Dot) then
            if time_now - cd_data.Flip_Time < 0.2 then cd_data.Flips = cd_data.Flips + 1 else cd_data.Flips = 0 end
            cd_data.Flip_Time = time_now
        end
        cd_data.Dot = dot
        cd_data.Lerp_Radians = cd_data.Lerp_Radians + (radians - cd_data.Lerp_Radians) * 0.8
        if speed > 100 and reach_time > ping_val / 10 then
            dist_thresh = math.max(dist_thresh - 15, 15)

        if dot < 0 then return true end
        if distance < dist_thresh then return false end

        if distance > dist_thresh and tick() - cd_data.Curving < reach_time / 1.5 then return true end

    return err

    elseif currentMode == "Right" then return (rightV * 150000 + (lookV * 0.001)).Unit
    elseif currentMode == "Up" then return (upV * 85000 + (lookV * 0.001)).Unit
    elseif currentMode == "Down" then return (-upV * 85000 + (lookV * 0.001)).Unit
    elseif currentMode == "Straight" then
        if hrp then return (hrp.CFrame.LookVector * 250000).Unit end
        return lookV
    elseif currentMode == "Random" then
        randomSpread = Vector3.new(math.random(-50000, 50000), math.random(-25000, 75000), math.random(-50000, 50000)).Unit
        return (lookV + (randomSpread * 350000)).Unit
    elseif currentMode == "Super Curve Random" then
        choices = {-rightV * 450000, rightV * 450000, upV * 250000, -upV * 250000}
        picked = choices[math.random(1, #choices)]
        return (picked + (lookV * 0.001)).Unit
    end
    xpcall(function()
        if ball and target and target.PrimaryPart then
            return ((target.PrimaryPart.Position - ball.Position).Unit * 500000).Unit
        end
    end, function(err) print("Curve dir calc failed: " .. tostring(err)) end)
    return lookV
end

end

FireParry = function(ball, isSpam)
    cam = workspace.CurrentCamera
    if not cam then return end
    target = ForceTarget()
    char = GetChar()
    hrp = GetHRP(char)
    alive = workspace:FindFirstChild("Alive")
    curveDirectionVector = CalcCurveDir(ball, target)
    finalCF = CFrame.lookAt(cam.CFrame.Position, cam.CFrame.Position + curveDirectionVector)
    snap = BuildSnap()
    gamePositionsSnapshot = {}
    if cam and alive then
        for _, character in ipairs(alive:GetChildren()) do
            if character:IsA("Model") and character.PrimaryPart then
                screenPos, onScreen = cam:WorldToScreenPoint(character.PrimaryPart.Position)
                gamePositionsSnapshot[character.Name] = screenPos
            end
        end
    end
    local packetSent = false
    if sendParry and not packetSent then
        local success, err = pcall(function()
            sendParry(finalCF, gamePositionsSnapshot, {cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2})
        end)
        if success then packetSent = true end
    end
    if _reverted and next(_reverted) ~= nil and not packetSent then
        local successCount = 0
        for remote, args in pairs(_reverted) do
            if type(args) ~= "table" then continue end
            local success, err = pcall(function()
                pa = {}
                pa[1] = args[1]; pa[2] = args[2]
                pa[3] = _tokenize and _tokenize(args[2]) or args[3]
                pa[4] = isSpam and 0 or 0.5
                pa[5] = finalCF; pa[6] = gamePositionsSnapshot
                pa[7] = {cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2}; pa[8] = false
                if remote:IsA("RemoteEvent") then remote:FireServer(unpack(pa))
                else remote:InvokeServer(unpack(pa)) end
            end)
            if success then successCount = successCount + 1 end
        end
        if successCount > 0 then packetSent = true end
    end
    if _firstSignal and not packetSent then
        local success, err = pcall(function()
            _firstSignal:Fire(0.5, finalCF, gamePositionsSnapshot, {cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2}, false)
        end)
        if success then packetSent = true end
    end
    if not packetSent then
        pcall(function()
            if mouse1press then mouse1press(); task.wait(0.03); mouse1release() end
            if mouse1click then mouse1click() end
        end)
    end
end

local function ParryNormal(ball, distance, speed, velocity, ping_ms)
    local endPosition   = 101 - math.clamp(Cfg.Accuracy, 1, 100)
    local maximumValue  = math.clamp((endPosition - 1) / 99, 0, 1)
    local offsetZ       = maximumValue * maximumValue * (3 - 2 * maximumValue)
    local openError     = math.clamp(ping_ms / 10, 10, 16.7)
    local step          = math.clamp(ping_ms / 1000 * 0.5 - 0.05, 0, 0.04)
    local clampedY      = 0  -- ParryRange fixed at 0
    local newDenom  = 2.4 + math.clamp(speed - 9.5, 0, 990.5) * 0.002
    local center    = speed / newDenom * (0.75 + maximumValue * 0.43)
    local isInside  = center <= 9.5
    local clampedX  = isInside
        and math.min(openError, speed * (ping_ms / 1000))
        or  openError + math.min(speed * step, 30)
    local zone = clampedX + math.max(center, 9.5) + clampedY
    if endPosition > 70 then
        local maxLen      = math.clamp((endPosition - 44) / 30, 0, 1)
        local index35     = maxLen * maxLen * (3 - 2 * maxLen)
        local index37     = (isInside and math.min(openError, speed * (ping_ms/1000)) or openError) + math.max(speed / newDenom * (0.75 + maximumValue * 0.2), 9.5)
        local index38     = openError + speed / newDenom * (1.4 + maximumValue * 0.2)
        local clampedDist = math.clamp((speed - 1000) / 800, 0, 1)
        local index39     = index38 + math.max(index37 - index38, 0) * clampedDist
        zone = zone + (index39 + (index37 - index39) * 0 - zone) * index35
    end
    return distance <= zone
end

local function CheckPhantomRelease(ball)
    if not PhantomTargetData or PhantomTargetData.Ball ~= ball then return false end
    if os.clock() >= PhantomTargetData.Expires then PhantomTargetData = nil return false end
    local enemy = PhantomTargetData.User
    if not enemy or not enemy.Parent then return false end
    if tostring(ball:GetAttribute("target") or "") ~= enemy.Name then
        PhantomTargetData.LastPos = nil; PhantomTargetData.Released = false; return false
    end
    local currentBallPos = GetBallPos(ball)
    if PhantomTargetData.LastPos and currentBallPos ~= PhantomTargetData.LastPos then
        PhantomTargetData.Released = true
    end
    PhantomTargetData.LastPos = currentBallPos
    return PhantomTargetData.Released == true
end

--=================================================================
-- MAIN PARRY LOOP
--=================================================================
local LastSingularityActiveTime = 0
local LastFrameTarget = {}
local _parryConn = nil

local function StartParryLoop()
    if _parryConn then _parryConn:Disconnect(); _parryConn = nil end
    local loopFn = function()
        xpcall(function()
            if not shared._IllusionRunning or not Cfg.AutoParry then return end
            local char = GetChar()
            local hrp = GetHRP(char)
            local hum = GetHum(char)
            if not hrp or not hum or hum.Health <= 0 then return end
            local alive = workspace:FindFirstChild("Alive")
            if not alive or not char:IsDescendantOf(alive) then return end
            local balls = GetBalls()
            if #balls == 0 then return end
            local myPos = hrp.Position
            local now = os.clock()

            -- Always use target-based selection (lobby parry removed from UI)
            local mainTargetBall = nil
            for _, b in ipairs(balls) do
                if b and b.Parent and b:GetAttribute("target") == LP.Name then
                    mainTargetBall = b; break
                end
            end

            if not mainTargetBall or not mainTargetBall.Parent then return end
            local ball = mainTargetBall
            local target = ball:GetAttribute("target")
            local isTargetingMe = (target == LP.Name)

            if parryLocked[ball] == true then return end

            if Cfg.CooldownProtection then
                local currentRunningTime = os.clock()
                for targetedBall, lockedTimestamp in pairs(lockTime) do
                    local currentTarget = targetedBall and targetedBall.Parent and targetedBall:GetAttribute("target")
                    local isTargetChanged = currentTarget and currentTarget ~= LP.Name
                    if parryLocked[targetedBall] == true and (isTargetChanged or (lockedTimestamp and lockedTimestamp > 0 and (currentRunningTime - lockedTimestamp) > 0.65)) then
                        parryLocked[targetedBall] = false
                        lockTime[targetedBall] = 0
                        if lastHitTick and lastHitTick[targetedBall] then lastHitTick[targetedBall] = 0 end
                        if activeParryTasks and activeParryTasks[targetedBall] then
                            task.cancel(activeParryTasks[targetedBall]); activeParryTasks[targetedBall] = nil
                        end
                    end
                end
            end

            local ballPos = GetBallPos(ball)
            local vel = GetBallVel(ball)
            local speed = vel.Magnitude
            local dist = (myPos - ballPos).Magnitude
            local ping = GetPing()
            local tti = dist / math.max(speed, 1)

            local isSingularActiveNow = (char:GetAttribute("IS_EVENT_SINGULARITY") == true)
            if isSingularActiveNow then LastSingularityActiveTime = now end
            if (now - LastSingularityActiveTime) <= 0.1 then
                if dist <= 12 and speed <= 8 then return end
            end

            local blocked = false
            pcall(function()
                if Cfg.InfinityDet and Det_State and Det_State.Infinity_Active then blocked = true end
                if Cfg.ForcefieldDet and Det_State and Det_State.Forcefield_Active then blocked = true end
                if Cfg.DeathSlashDet and Det_State and Det_State.DeathSlash_Active then blocked = true end
                if Cfg.AntiPhantom and PhantomTargetData and not PhantomTargetData.Released then blocked = true end
            end)
            if blocked then return end

            local curved = Cfg.AntiCurve and (IsCurved(ball) or IsCurved2(ball))
            if curved then return end

            local trigger = false
            if speed >= 795 and Cfg.ProtectionFastBall then
                local pingThresh = math.clamp(ping, 5, 17)
                local skyBuf = vel.Y < -5 and math.max(math.abs(vel.Y) * 0.42, 12) or 0
                local uScale = (Cfg.DistanceTiming or 100) / 100
                local direction = (myPos - ballPos).Unit
                local dotProduct = direction:Dot(vel.Unit)
                if (dist <= ((pingThresh + math.max(speed / (2.4 + math.min(math.max(speed - 9.5, 0), 650) * 0.002), 9.5)) * uScale * 1.18) + skyBuf or tti <= 0.065) and (dotProduct > 0.35 or dist <= 35 + skyBuf) then
                    trigger = true
                end
            end

            if not trigger then
                if dist <= 5.5 then
                    trigger = true
                elseif Cfg.ParryMode == "Normal" then
                    trigger = ParryNormal(ball, dist, speed, vel, ping)
                elseif Cfg.ParryMode == "Advance" then
                    local cap = math.min(math.max(speed - 9.5, 0), 650)
                    local div = (2.4 + cap * 0.002) * (Cfg.Accuracy and (0.7 + (Cfg.Accuracy - 1) * (0.9 / 99)) or 1.1)
                    local currentPingMs = ping
                    local zone = math.clamp(currentPingMs / 10, 5, 17) + math.max(speed / div, 9.5)
                    if speed >= 450 then
                        zone = zone + (speed / 500) * math.clamp(currentPingMs / 10, 4, 11)
                    end
                    local curvedVirex = IsCurved_virex(ball)
                    if dist <= 14.5 then trigger = true
                    elseif dist <= zone and not curvedVirex then trigger = true end
                end
            end

            local tornado = workspace:FindFirstChild("Runtime") and workspace.Runtime:FindFirstChild("Tornado")
            local tornadoTime = tornado and (tornado:GetAttribute("TornadoTime") or 1)
            local isAerodynamic = ball:GetAttribute("AerodynamicTime") or 0

            if isTargetingMe then
                if LastFrameTarget[ball] == LP.Name and tornado and (now - isAerodynamic) < (tornadoTime + 0.314159) then
                    trigger = false
                    parryLocked[ball] = true
                    task.delay(0.12, function()
                        xpcall(function()
                            if ball and ball.Parent then
                                if tostring(ball:GetAttribute("target") or "") == LP.Name then
                                    parryLocked[ball] = false
                                    FireParry(ball, false)
                                end
                            end
                        end, function(err) print("Tornado counter error: " .. tostring(err)) end)
                    end)
                end
            end

            LastFrameTarget[ball] = target

            if trigger then
                if not isTargetingMe then return end
                local now = os.clock()
                local fst = getgenv().FlyteBB_State
                if fst then
                    if fst.ParryCooldownUntil and now < fst.ParryCooldownUntil then return end
                    if fst.ExtraParryCooldownUntil and now < fst.ExtraParryCooldownUntil then return end
                    fst.ExtraParryCooldownUntil = nil
                end
                local pingS = ping / 1000
                local minWindow = math.clamp(pingS + 0.1, 0.15, 1)
                local recentlySent = fst and fst.AutoParryAt and (now - fst.AutoParryAt) <= minWindow
                if not recentlySent and (now - lastParryTime) >= 0.005 then
                    parryLocked[ball] = true
                    lockTime[ball]    = now
                    FireParry(ball, false)
                    lastParryTime = now
                    if fst then fst.AutoParryAt = now end
                end
            end
        end, function(e) print("Nexora Main Loop Error: " .. tostring(e)) end)
    end

    local loop = Cfg.ParryLoop
    if loop == "PreRender" then _parryConn = RunService.PreRender:Connect(loopFn)
    elseif loop == "Heartbeat" then _parryConn = RunService.Heartbeat:Connect(loopFn)
    else _parryConn = RunService.PreSimulation:Connect(loopFn) end
end

local _spamConn = nil

local function AutoSpamLoop()
    if not Cfg.AutoSpam then shared.ManualSpamActive = false; return end
    local char = GetChar()
    local hrp  = GetHRP(char)
    local hum  = GetHum(char)
    if not hrp or not hum or hum.Health <= 0 then shared.ManualSpamActive = false; return end
    if char:GetAttribute("Pulsed") or (Det_State and Det_State.DeathSlash_Active) then
        shared.ManualSpamActive = false; return
    end
    local myPos = hrp.Position
    local balls = GetBalls()
    if #balls == 0 then shared.ManualSpamActive = false; return end
    local mainBall = nil
    for _, b in ipairs(balls) do
        if b and b.Parent then mainBall = b; break end
    end
    if not mainBall then shared.ManualSpamActive = false; return end
    local ballPos = GetBallPos(mainBall)
    local vel     = GetBallVel(mainBall)
    local speed   = vel.Magnitude
    local ballDist = (myPos - ballPos).Magnitude
    if speed < 1 or ballDist < 0.1 then shared.ManualSpamActive = false; return end
    local alive = workspace:FindFirstChild("Alive")
    local currentBallTarget = tostring(mainBall:GetAttribute("target") or mainBall:GetAttribute("Target") or "")
    local fromAttr = currentBallTarget == LP.Name and mainBall:GetAttribute("from") or currentBallTarget
    local enemyChar = type(fromAttr) == "string" and fromAttr ~= "" and alive and alive:FindFirstChild(fromAttr)
    local enemyHRP = enemyChar and (enemyChar.PrimaryPart or enemyChar:FindFirstChild("HumanoidRootPart"))
    local dynamicMaxDistance = tonumber(Cfg.SpamThreshold) or 35
    if Cfg.autospamrange then
        dynamicMaxDistance = math.clamp(10 + (speed * 0.3), 15, 60)
    end
    local distanceToEnemy = enemyHRP and (myPos - enemyHRP.Position).Magnitude or math.huge
    if distanceToEnemy <= dynamicMaxDistance then
        if currentBallTarget == LP.Name or currentBallTarget == tostring(fromAttr) then
            shared.ManualSpamActive = true
            local burstCount = (speed > 1800) and 3 or 1
            if Cfg.DynamicSpam then
                burstCount = math.clamp(math.ceil((dynamicMaxDistance / math.max(ballDist, 1)) * 1.5), 1, 4)
            end
            for i = 1, burstCount do
                pcall(FireParry, mainBall, true)
                pcall(FireParry, mainBall, true)
            end
        else
            shared.ManualSpamActive = false
        end
    else
        shared.ManualSpamActive = false
    end
end

local _tbConn = nil
local _lastTbTick = 0

local function SetTriggerbot(on)
    if _tbConn then _tbConn:Disconnect(); _tbConn = nil end
    if not on then return end
    _tbConn = RunService.Heartbeat:Connect(function()
        xpcall(function()
            if not Cfg.TriggerBot then return end
            local ball = GetBall()
            if not ball or not ball.Parent then return end
            local targetAttr = ball:GetAttribute("target") or ball:GetAttribute("Target")
            if tostring(targetAttr or "") ~= LP.Name then return end
            if parryLocked[ball] then return end
            local now = os.clock()
            if (now - _lastTbTick) >= 0.12 then
                _lastTbTick = now
                parryLocked[ball] = true; lockTime[ball] = now
                FireParry(ball, false)
            end
        end, function(err) print("Triggerbot error: " .. tostring(err)) end)
    end)
end

local _flyConn, _flyBV, _flyGyro = nil, nil, nil

local function StartFly()
    local char = GetChar()
    local hrp = GetHRP(char)
    local hum = GetHum(char)
    if not hrp or not hum then return end
    hum.PlatformStand = true
    if _flyBV then _flyBV:Destroy() end
    if _flyGyro then _flyGyro:Destroy() end
    if _flyConn then _flyConn:Disconnect() end
    _flyBV = Instance.new("BodyVelocity")
    _flyBV.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    _flyBV.Velocity = Vector3.zero
    _flyBV.Parent = hrp
    _flyGyro = Instance.new("BodyGyro")
    _flyGyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    _flyGyro.P = 9e4
    _flyGyro.Parent = hrp
    _flyConn = RunService.RenderStepped:Connect(function()
        if not Cfg.FlyEnabled or not hrp or not hum or not _flyBV or not _flyGyro then return end
        hum.PlatformStand = true
        local cam = workspace.CurrentCamera
        if not cam then return end
        _flyGyro.CFrame = cam.CFrame
        local moveVector = Vector3.zero
        local controlModule = nil
        pcall(function()
            controlModule = require(LP.PlayerScripts:FindFirstChild("PlayerModule"):FindFirstChild("ControlModule"))
        end)
        if controlModule and controlModule.GetMoveVector then
            local rawVector = controlModule:GetMoveVector()
            moveVector = Vector3.new(rawVector.X, 0, rawVector.Z)
        else
            moveVector = hum.MoveDirection
        end
        if moveVector.Magnitude == 0 then
            if UIS:IsKeyDown(Enum.KeyCode.W) then moveVector = moveVector + Vector3.new(0, 0, -1) end
            if UIS:IsKeyDown(Enum.KeyCode.S) then moveVector = moveVector + Vector3.new(0, 0, 1) end
            if UIS:IsKeyDown(Enum.KeyCode.A) then moveVector = moveVector + Vector3.new(-1, 0, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.D) then moveVector = moveVector + Vector3.new(1, 0, 0) end
        end
        local sp = Cfg.FlySpeed or 50
        local direction = cam.CFrame:VectorToWorldSpace(moveVector)
        local verticalVel = 0
        if UIS:IsKeyDown(Enum.KeyCode.Space) then verticalVel = sp end
        if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then verticalVel = -sp end
        _flyBV.Velocity = (direction * sp) + Vector3.new(0, verticalVel, 0)
    end)
end

local function StopFly()
    if _flyConn then _flyConn:Disconnect(); _flyConn = nil end
    if _flyBV then _flyBV:Destroy(); _flyBV = nil end
    if _flyGyro then _flyGyro:Destroy(); _flyGyro = nil end
    local hum = GetHum(GetChar())
    if hum then hum.PlatformStand = false end
end

end

local ImmortalCamAnchor = Instance.new("Part")
ImmortalCamAnchor.Size = Vector3.new(0.1, 0.1, 0.1)

ImmortalCamAnchor.CanCollide = false
ImmortalCamAnchor.Anchored = true

RunService.Heartbeat:Connect(function()
    local char = GetChar(); local hrp = GetHRP(char); local hum = GetHum(char)
    local cam  = workspace.CurrentCamera
    if not hrp or not hum or hum.Health <= 0 or not cam then
        if cam and cam.CameraSubject == ImmortalCamAnchor then cam.CameraSubject = hum end
        return
    end
    if Cfg.ImmortalV1 or Cfg.ImmortalV2 then
        ImmortalCamAnchor.CFrame = ImmortalCamAnchor.CFrame:Lerp(CFrame.new(hrp.Position), 0.3)
        if cam.CameraSubject ~= ImmortalCamAnchor then cam.CameraSubject = ImmortalCamAnchor end
        pcall(function() setfflag("S2PhysicsSenderRate", "15") end)
    else
        if cam.CameraSubject == ImmortalCamAnchor then cam.CameraSubject = hum end
    end
    if Cfg.ImmortalV1 then
        local t = tick(); local sp = math.floor(t * 95) % 3; local r = 14
        hrp.CFrame = hrp.CFrame + Vector3.new(0, 0.01, 0)
        local lv = hrp.CFrame.LookVector; local rv = hrp.CFrame.RightVector
        local tri, vel = Vector3.zero, Vector3.zero
        if sp == 0 then tri = (-rv*r)+(lv*(r*0.5)); vel = -rv*350
        elseif sp == 1 then tri = (rv*r)+(lv*(r*0.5)); vel = rv*350
        else tri = -lv*r; vel = -lv*350 end
        local oCF, oVel = hrp.CFrame, hrp.AssemblyLinearVelocity
        hrp.CFrame = oCF * CFrame.new(tri); hrp.AssemblyLinearVelocity = vel
        RunService.RenderStepped:Wait()
        if hrp and hrp.Parent then hrp.CFrame = oCF; hrp.AssemblyLinearVelocity = oVel end
    elseif Cfg.ImmortalV2 then
        local ang = tonumber(Cfg.ImmortalAngle) or 145
        local t = tick()
        hrp.CFrame = hrp.CFrame + Vector3.new(0, 0.01, 0)
        local ca = t * math.pi * 2 * ang / 5
        local y = (math.floor(t*135) % 2 == 0) and -8 or 35
        local rad = tonumber(Cfg.ImmortalRadius) or 10
        local off = Vector3.new(math.cos(ca)*rad, y, math.sin(ca)*rad)
        local oCF, oVel = hrp.CFrame, hrp.AssemblyLinearVelocity
        hrp.CFrame = oCF + off
        hrp.AssemblyLinearVelocity = Vector3.new(math.sin(ca)*250, y == 35 and 280 or -280, math.cos(ca)*250)
        RunService.RenderStepped:Wait()
        if hrp and hrp.Parent then hrp.CFrame = oCF; hrp.AssemblyLinearVelocity = oVel end
    end
end)

    pcall(function()
        local c = GetChar(); if not c then return end

            h.Transparency = on and 1 or 0
            for _, d in pairs(h:GetChildren()) do
                if d:IsA("Decal") then d.Transparency = on and 1 or 0 end
            end
        end
        for _, acc in pairs(c:GetChildren()) do
            if acc:IsA("Accessory") then

                if hnd then
                    local hrp = GetHRP(c)
                    if hrp and (hnd.Position - hrp.Position).Y > 0.8 then

                    end
                end
            end
        end
    end)
end

-- AUTO PLAY (improved: smarter movement, handles ball approach)
--=================================================================
task.spawn(function()
    local lastBallDist = math.huge
    while shared._IllusionRunning do
        task.wait(math.random(2, 4))
        if not Cfg.AutoPlay then lastBallDist = math.huge; continue end
        local char  = GetChar()
        local alive = workspace:FindFirstChild("Alive")
        if not char or not alive or not char:IsDescendantOf(alive) then continue end
        local hum  = GetHum(char)
        local hrp  = GetHRP(char)
        if not hum or not hrp then continue end
        pcall(function()
            local ball = GetBall()
            local myPos = hrp.Position

            if ball and ball.Parent then
                local ballPos = GetBallPos(ball)
                local ballTarget = ball:GetAttribute("target")
                local vel = GetBallVel(ball)
                local dist = (myPos - ballPos).Magnitude

                -- If ball is coming toward us fast, step sideways to reposition
                if ballTarget == LP.Name and vel.Magnitude > 50 and dist < 40 then
                    local sideDir = hrp.CFrame.RightVector * (math.random(0, 1) == 0 and 8 or -8)
                    hum:MoveTo(myPos + sideDir)
                    return
                end

                -- If ball is far away and not targeting us, walk near the center
                if dist > 80 then
                    local center = Vector3.new(0, myPos.Y, 0)
                    hum:MoveTo(center + Vector3.new(math.random(-20, 20), 0, math.random(-20, 20)))
                    return
                end
            end

            -- Default: random roam in reasonable radius
            local ang = math.random() * math.pi * 2
            local radius = math.random(8, 25)
            hum:MoveTo(hrp.Position + Vector3.new(math.cos(ang) * radius, 0, math.sin(ang) * radius))
        end)
        -- Occasional jump to look alive
        if math.random() < 0.15 then
            pcall(function()
                local hum = GetHum(GetChar())
                if hum then hum.Jump = true end
            end)
        end
    end
end)

            if ok and conns then

                    if v.Function and islclosure and islclosure(v.Function) then

                        if ok2 and #up == 1 and type(up[1]) == "table" then
                            swordsCtrl = up[1]; break
                        end
                    end
                end
            end
        end
    end)

        local ok, res = pcall(function() return Swords:GetSword(name) end)

        local targetFX = getgenv().swordFX ~= "" and getgenv().swordFX or getgenv().swordModel
        getgenv().slashName = targetFX ~= "" and getSlashName(targetFX) or "SlashEffect"

        if not getgenv().skinChanger then return end
        local char = GetChar(); if not char then return end

            local equipFunc = rawget(Swords, "EquipSwordTo")
            if type(equipFunc) == "function" then
                local ups = getupvalues(equipFunc)
                for i = 1, #ups do
                    if type(ups[i]) == "boolean" then setupvalue(equipFunc, i, false); break end
                end
            end
        end)
        pcall(function() Swords:EquipSwordTo(char, getgenv().swordModel) end)

            while not swordsCtrl do task.wait(0.5) end
            pcall(function()

            pcall(function()

                                pcall(orig, unpack(args))
                            end)
                        end
                    end
                end
            end
        end
    end)

local function ApplyAvatarMorph(targetInput)
    if not targetInput or targetInput == "" then return end
    local character = game:GetService("Players").LocalPlayer.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local head = character:FindFirstChild("Head")
    if not humanoid or not head then return end
    local num = tonumber(targetInput)
    if not num then pcall(function() num = game:GetService("Players"):GetUserIdFromNameAsync(targetInput) end) end
    if not num then return end
    local ok, result = pcall(function() return game:GetService("Players"):CreateHumanoidModelFromUserId(num) end)
    if ok and result then
        local head2 = result:FindFirstChild("Head")
        local humanoid2 = result:FindFirstChildOfClass("Humanoid")
        if head2 then
            local decal = head:FindFirstChildOfClass("Decal")
            local decal2 = head2:FindFirstChildOfClass("Decal")
            if decal2 then
                if not decal then decal = Instance.new("Decal"); decal.Name = "face"; decal.Parent = head end
                decal.Texture = decal2.Texture
            elseif decal then decal:Destroy() end
            local specialMesh = head:FindFirstChildOfClass("SpecialMesh")
            local specialMesh2 = head2:FindFirstChildOfClass("SpecialMesh")
            if specialMesh2 then
                if not specialMesh then specialMesh = Instance.new("SpecialMesh"); specialMesh.Parent = head end
                specialMesh.MeshId = specialMesh2.MeshId
                specialMesh.TextureId = specialMesh2.TextureId
                specialMesh.Scale = specialMesh2.Scale
            end
        end
        for _, child in ipairs(character:GetChildren()) do
            if child:IsA("Accessory") or child:IsA("Clothing") or child:IsA("ShirtGraphic") or child:IsA("BodyColors") or child:IsA("CharacterMesh") then
                child:Destroy()
            end
        end
        local bodyColors = result:FindFirstChildOfClass("BodyColors")
        if bodyColors then bodyColors:Clone().Parent = character end
        if humanoid.RigType == Enum.HumanoidRigType.R15 and humanoid2 then
            for _, v90 in ipairs({"BodyHeightScale","BodyWidthScale","BodyDepthScale","HeadScale","ProportionScale"}) do
                local v91 = humanoid2:FindFirstChild(v90); local v92 = humanoid:FindFirstChild(v90)
                if v91 and v92 then v92.Value = v91.Value end
            end
        end
        if humanoid.RigType == Enum.HumanoidRigType.R6 then
            for _, child in ipairs(result:GetChildren()) do
                if child:IsA("CharacterMesh") then child:Clone().Parent = character end
            end
        else
            for _, child in ipairs(result:GetChildren()) do
                if child:IsA("MeshPart") then
                    local v90 = character:FindFirstChild(child.Name)
                    if v90 and v90:IsA("MeshPart") then v90.MeshId = child.MeshId; v90.TextureID = child.TextureID end
                end
            end
        end
        for _, child in ipairs(result:GetChildren()) do
            if child:IsA("Accessory") then
                local clone = child:Clone(); clone.Parent = character
                local handle = clone:FindFirstChild("Handle")
                if handle and handle:IsA("BasePart") then
                    handle.CanCollide = false
                    if not handle:FindFirstChildOfClass("WrapLayer") then
                        local attachment = handle:FindFirstChildOfClass("Attachment")
                        local v90 = nil
                        if attachment then
                            for _, child2 in ipairs(character:GetChildren()) do
                                if child2:IsA("BasePart") and child2:FindFirstChild(attachment.Name) then
                                    v90 = child2:FindFirstChild(attachment.Name); break
                                end
                            end
                        end
                        if v90 then
                            handle.CFrame = v90.WorldCFrame * attachment.CFrame:Inverse()
                            local wc = Instance.new("WeldConstraint"); wc.Part0 = handle; wc.Part1 = v90.Parent; wc.Parent = handle
                        else
                            local torso = head or character:FindFirstChild("Torso") or character:FindFirstChild("UpperTorso")
                            if torso then
                                handle.CFrame = torso.CFrame
                                local wc = Instance.new("WeldConstraint"); wc.Part0 = handle; wc.Part1 = torso; wc.Parent = handle
                            end
                        end
                    end
                end
            elseif child:IsA("Clothing") or child:IsA("ShirtGraphic") then
                child:Clone().Parent = character
            end
        end
        result:Destroy()
    end
end

            local pg=UA.Shop._inventoryPages and UA.Shop._inventoryPages[t]
            if pg then onInputEnded("MarkDirty"..t,pg,"MarkDirty") end
        end
        cleanConn(UA.Connections); table.clear(UA.Connections)
        cleanConn(UA.CardConnections); table.clear(UA.CardConnections)
        UA.RestoreInventory(restoreAll); UA.RestoreShop(); UA.RestoreFavorites()
        LP:SetAttribute("ShowSwordAccessory",UA.Defaults.Accessory==true)
        for k,def in pairs(UA.DefaultFinishers) do
            if UA.DataReplion then UA.DataReplion:_set({"Finishers","Equipped",k},def==true) end
        end
        table.clear(UA.Finishers); table.clear(UA.DefaultFinishers)
        onInputEnded("PlayExplosion",UA.VFX); onInputEnded("EmotePlay",UA.EmoteController,"Play")
        onInputEnded("UpdateHolderEmotes",UA.EmoteWheel); onInputEnded("UpdateMenuContentEmotes",UA.EmoteWheel)
        UA.RestoreSword(); UA.RestoreExplosion(); UA.RestoreEmotes()
        if UA.Originals.SetSword and UA.BladeController then onInputEnded("SetSword",UA.BladeController) end
    end
    UA.Start=function()
        if UA.Status then return end; UA.Status="starting"; local RUN=UA.Run+1; UA.Run=RUN
        if not UA.Load() then UA.Status=nil; return end
        while UA._enabled and UA.Run==RUN and not UA.ShopReady() do RunService.Heartbeat:Wait() end
        if not UA._enabled or UA.Run~=RUN then if UA.Run==RUN then UA.Status=nil end; return end
        UA.Status="active"
        for _,d in pairs(UA.Deleted) do table.clear(d) end
        table.clear(UA.EquippedEmotes); table.clear(UA.SavedEquipped)
        table.clear(UA.Finishers); table.clear(UA.DefaultFinishers)
        UA.Accessory=false; UA.Originals.EquippedSword=nil; UA.Originals.EquippedExplosion=nil
        UA.LoadFavorites(); UA.SaveDefaults()
        local inv=UA.LoadEmotes(); if UA.Run~=RUN or UA.Status~="active" then return end
        UA.PatchInventory(inv); UA.HookEmotes()
        UA.PatchExplosion(); UA.PatchEmotes(); UA.PatchParrySuccess()

        end))
        UA.Add(LP.CharacterAdded:Connect(function(char)

            if UA.WatchEmotes then UA.WatchEmotes(char) end
            task.delay(1,function()

                if UA.Explosion then LP:SetAttribute("CurrentlyEquippedExplosion",UA.Explosion); char:SetAttribute("CurrentlyEquippedExplosion",UA.Explosion) end
            end)
        end))

        if UA.Accessory and UA.Sword then UA.ApplySword(UA.Sword)

        end

    end
    UA.WatchEmotes=function(char)

        local hum=char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid",2); if not hum then return end
        UA.Add(hum:GetPropertyChangedSignal("MoveDirection"):Connect(function()
            if hum.MoveDirection~=Vector3.zero and UA.EmoteTrove then UA.StopEmote() end
        end))
    end
end

local function makeDraggable(handle, target)
    local dragging, start, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging, start, startPos = true, i.Position, target.Position
            target:SetAttribute("moved", false)
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    bind(UIS.InputChanged, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - start
            if d.Magnitude > 5 then target:SetAttribute("moved", true) end
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset+d.X, startPos.Y.Scale, startPos.Y.Offset+d.Y)
        end
    end)
end

local function Toggle(page, lbl, desc, default, cb)
    local cfgKey = lbl:gsub("%s+","")
    local val = default or false
    if Cfg[cfgKey] ~= nil then val = Cfg[cfgKey] end
    local Row = new("Frame", {Size=UDim2.new(1,0,0,desc and 44 or 32), BackgroundColor3=C.PANEL2, BackgroundTransparency=0.35, BorderSizePixel=0, Parent=NPages[page]}, {corner(8), stroke(C.PURPLE,1,0.78)})
    local L = label(Row, {Position=UDim2.fromOffset(10,desc and 4 or 0), Size=UDim2.new(1,-65,0,16), Text=lbl, TextSize=12, Font=Enum.Font.GothamMedium})
    if desc then label(Row, {Position=UDim2.fromOffset(10,22), Size=UDim2.new(1,-65,0,14), Text=desc, TextSize=9, TextColor3=C.MUTED}) end
    local Tr = new("Frame", {Size=UDim2.fromOffset(32,16), Position=UDim2.new(1,-42,0.5,-8), BackgroundColor3=val and C.GREEN or C.OFF, Parent=Row}, {corner(20)})
    local Kn = new("Frame", {Size=UDim2.fromOffset(10,10), Position=val and UDim2.new(1,-13,0.5,-5) or UDim2.fromOffset(3,3), BackgroundColor3=WHITE, Parent=Tr}, {corner(20)})
    local TB = new("TextButton", {Size=UDim2.fromScale(1,1), BackgroundTransparency=1, Text="", Parent=Tr})
    TB.MouseButton1Click:Connect(function()
        val = not val
        tw(Tr,0.14,{BackgroundColor3=val and C.GREEN or C.OFF})
        tw(Kn,0.14,{Position=val and UDim2.new(1,-13,0.5,-5) or UDim2.fromOffset(3,3)})
        Cfg[cfgKey] = val
        xpcall(function() if cb then cb(val) end end, function(err) print("Toggle cb error: "..tostring(err)) end)
        SaveConfig()
    end)
    Row.MouseEnter:Connect(function() tw(Row,0.08,{BackgroundTransparency=0.2}) end)
    Row.MouseLeave:Connect(function() tw(Row,0.08,{BackgroundTransparency=0.35}) end)
    return {
        Set=function(_,v)
            val=v; Cfg[cfgKey]=v
            tw(Tr,0.14,{BackgroundColor3=v and C.GREEN or C.OFF})
            tw(Kn,0.14,{Position=v and UDim2.new(1,-13,0.5,-5) or UDim2.fromOffset(3,3)})
            xpcall(function() if cb then cb(v) end end, function() end); SaveConfig()
        end,
        Get=function() return val end
    }
end

local function Select(page, lbl, opts, cb)
    local cfgKey = lbl:gsub("%s+","")
    local sel = opts[1]; if Cfg[cfgKey] ~= nil then sel = Cfg[cfgKey] end
    local H = new("Frame", {Size=UDim2.new(1,0,0,40), BackgroundColor3=C.PANEL2, BackgroundTransparency=0.35, BorderSizePixel=0, Parent=NPages[page]}, {corner(8), stroke(C.PURPLE,1,0.78)})
    label(H, {Position=UDim2.fromOffset(12,0), Size=UDim2.new(1,-160,1,0), Text=lbl, TextSize=12, Font=Enum.Font.GothamMedium})
    local SB = button(H, {Size=UDim2.fromOffset(140,28), Position=UDim2.new(1,-150,0.5,-14), BackgroundColor3=C.PANEL2, BackgroundTransparency=0.2, Text=tostring(sel).." ▾", TextSize=11, Font=Enum.Font.GothamMedium, TextColor3=C.TEXT}, {corner(8), stroke(C.PURPLE,1,0.6)})
    local Drop = new("Frame", {BackgroundColor3=C.PANEL, BorderSizePixel=0, Size=UDim2.fromOffset(140,0), Visible=false, ZIndex=50, Parent=NPages[page]}, {corner(8), stroke(C.PURPLE,1.2,0)})
    new("UIListLayout", {Padding=UDim.new(0,2), Parent=Drop})
    new("UIPadding", {PaddingTop=UDim.new(0,4),PaddingLeft=UDim.new(0,4),PaddingRight=UDim.new(0,4),PaddingBottom=UDim.new(0,4), Parent=Drop})
    for _, opt in ipairs(opts) do
        local OB = button(Drop, {Size=UDim2.new(1,0,0,28), BackgroundColor3=C.PANEL2, BackgroundTransparency=0.5, Text=tostring(opt), TextSize=11, Font=Enum.Font.GothamMedium, TextColor3=C.TEXT, ZIndex=51}, {corner(6)})
        OB.MouseEnter:Connect(function() tw(OB,0.08,{BackgroundTransparency=0.1}) end)
        OB.MouseLeave:Connect(function() tw(OB,0.08,{BackgroundTransparency=0.5}) end)
        OB.MouseButton1Click:Connect(function()
            sel=opt; SB.Text=tostring(opt).." ▾"
            tw(Drop,0.1,{Size=UDim2.fromOffset(140,0)}); task.delay(0.1, function() Drop.Visible=false end)
            xpcall(function() if cb then cb(opt) end end, function(err) print("Select cb: "..tostring(err)) end)
            Cfg[cfgKey]=opt; SaveConfig()
        end)
    end
    SB.MouseButton1Click:Connect(function()
        if Drop.Visible then
            tw(Drop,0.1,{Size=UDim2.fromOffset(140,0)}); task.delay(0.1, function() Drop.Visible=false end)
        else
            local abs=SB.AbsolutePosition; local sz=SB.AbsoluteSize; local pageAbs=NPages[page].AbsolutePosition
            Drop.Position=UDim2.fromOffset(abs.X-pageAbs.X, (abs.Y-pageAbs.Y)+sz.Y+3)
            Drop.Size=UDim2.fromOffset(140,0); Drop.Visible=true
            tw(Drop,0.15,{Size=UDim2.fromOffset(140,math.min(#opts*32+8,175))})
        end
    end)
    NPages[page]:GetPropertyChangedSignal("Visible"):Connect(function()
        if not NPages[page].Visible then Drop.Visible=false end
    end)
    return {Get=function() return sel end, Set=function(_,v) sel=v; SB.Text=tostring(v).." ▾"; xpcall(function() if cb then cb(v) end end, function() end) end}
end

local function Slider(page, lbl, mn, mx, def, fmt, cb)
    local cfgKey = lbl:gsub("%s+","")
    if lbl=="Cooldown Value ×100" then cfgKey="CooldownValue" end
    local val=def; if Cfg[cfgKey]~=nil then
        val=(lbl=="Cooldown Value ×100") and Cfg[cfgKey]*100 or Cfg[cfgKey]
    end
    local H=new("Frame",{Size=UDim2.new(1,0,0,58),BackgroundColor3=C.PANEL2,BackgroundTransparency=0.35,BorderSizePixel=0,Parent=NPages[page]},{corner(8),stroke(C.PURPLE,1,0.78)})
    label(H,{Position=UDim2.fromOffset(12,6),Size=UDim2.new(1,-70,0,16),Text=lbl,TextSize=12,Font=Enum.Font.GothamMedium})
    local VL=label(H,{AnchorPoint=Vector2.new(1,0),Position=UDim2.new(1,-14,0,6),Size=UDim2.fromOffset(48,16),Text=(fmt or "%d"):format(val),TextSize=11,TextColor3=C.RED,TextXAlignment=Enum.TextXAlignment.Right})
    local TBG=new("Frame",{Size=UDim2.new(1,-24,0,5),Position=UDim2.fromOffset(12,35),BackgroundColor3=C.OFF,BorderSizePixel=0,Parent=H},{corner(4)})
    local Fill=new("Frame",{BackgroundColor3=WHITE,BorderSizePixel=0,Parent=TBG},{corner(4),tri(C.RED,C.PURPLE,C.BLUE,0)})
    local Knob=new("Frame",{Size=UDim2.fromOffset(11,11),AnchorPoint=Vector2.new(0.5,0.5),BackgroundColor3=WHITE,BorderSizePixel=0,ZIndex=3,Parent=TBG},{corner(7),stroke(C.PURPLE,2,0.1)})
    local function UpdateVisuals(v2)
        local p=math.clamp((v2-mn)/(mx-mn),0,1)
        tw(Fill,0.1,{Size=UDim2.fromScale(p,1)}); tw(Knob,0.1,{Position=UDim2.fromScale(p,0.5)})
        VL.Text=(fmt or "%d"):format(v2)
    end
    UpdateVisuals(val)
    local drg=false
    local function Upd(px)
        xpcall(function()
            local p=math.clamp((px-TBG.AbsolutePosition.X)/math.max(TBG.AbsoluteSize.X,1),0,1)
            val=math.floor(mn+(mx-mn)*p)
            Cfg[cfgKey]=(lbl=="Cooldown Value ×100") and val/100 or val
            UpdateVisuals(val); if cb then cb(val) end
        end, function() end)
    end
    TBG.InputBegan:Connect(function(i) if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then drg=true; Upd(i.Position.X) end end)
    bind(UIS.InputChanged, function(i) if drg and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then Upd(i.Position.X) end end)
    bind(UIS.InputEnded, function(i) if drg and (i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch) then drg=false; SaveConfig() end end)
    return {Get=function() return val end, Set=function(_,v) val=v; Cfg[cfgKey]=v; UpdateVisuals(v); if cb then cb(v) end; SaveConfig() end}
end

    end)

    tw(mscale,0.35,{Scale=1},Enum.EasingStyle.Back)

end)
