            data.First_Hit = data.First_Hit or from

    data.TargetConn = ball:GetAttributeChangedSignal("target"):Connect(function()
        parryLocked[ball] = false
        data.Target_Count  = (data.Target_Count or 0) + 1
    end)

local function ForceTarget()
    local alive = workspace:FindFirstChild("Alive")
    local char  = GetChar()
    local hrp   = GetHRP(char)
    if not alive or not hrp then return nil end
    if Cfg.TargetMode == "Double Click" and ForceTargetPlayer then
        local tc = alive:FindFirstChild(ForceTargetPlayer)
        if tc and tc ~= char then return tc end
    end
    if Cfg.TargetMode == "Nearest" then
        local best, bestD = nil, math.huge
        for _, c2 in ipairs(alive:GetChildren()) do
            if c2 ~= char and c2.PrimaryPart then

                if d < bestD then bestD = d; best = c2 end
            end
        end
        return best
    end
    return nil
end

-- Curve direction calculation
function CalcCurveDir(ball, target)
    camera = workspace.CurrentCamera
    char = game.Players.LocalPlayer.Character
    hrp = char and char:FindFirstChild("HumanoidRootPart")
    currentMode = Cfg.CurveMode or "Camera"
    lookV = camera.CFrame.LookVector
    upV = camera.CFrame.UpVector
    rightV = camera.CFrame.RightVector
    
    if currentMode == "Camera" then
        return lookV
    elseif currentMode == "Left" then
        return (-rightV * 150000 + (lookV * 0.001)).Unit
    elseif currentMode == "Right" then
        return (rightV * 150000 + (lookV * 0.001)).Unit
    elseif currentMode == "Up" then
        return (upV * 85000 + (lookV * 0.001)).Unit
    elseif currentMode == "Down" then
        return (-upV * 85000 + (lookV * 0.001)).Unit
    elseif currentMode == "Straight" then
        if hrp then
            return (hrp.CFrame.LookVector * 250000).Unit
        end
        return lookV
    elseif currentMode == "Random" then
        randomSpread = Vector3.new(math.random(-50000, 50000), math.random(-25000, 75000), math.random(-50000, 50000)).Unit
        return (lookV + (randomSpread * 350000)).Unit
    elseif currentMode == "Super Curve Random" then
        choices = {-rightV * 450000, rightV * 450000, upV * 250000, -upV * 250000}
        picked = choices[math.random(1, #choices)]
        return (picked + (lookV * 0.001)).Unit
    end
    
    xpcall(function()
        if ball and target and target.PrimaryPart then
            return ((target.PrimaryPart.Position - ball.Position).Unit * 500000).Unit
        end
    end, function(err)
        print("Illusion Engine Error: Target curve trajectory extreme calculation failed due to " .. tostring(err))
    end)
    
    return lookV
end

-- Build snap map
local function BuildSnap()
    local cam   = workspace.CurrentCamera
    local snap  = {}
    local alive = workspace:FindFirstChild("Alive")
    if alive and cam then
        for _, c2 in ipairs(alive:GetChildren()) do
            local p2 = c2.PrimaryPart or c2:FindFirstChild("HumanoidRootPart")
            if p2 then
                local sp = cam:WorldToScreenPoint(p2.Position)
                if sp.Z > 0 then snap[c2.Name] = sp end
            end
        end
    end
    return snap
end

FireParry = function(ball, isSpam)
    cam = workspace.CurrentCamera
    if not cam then 
        print("Illusion Engine Error: CurrentCamera object is not found")
        return 
    end
    target = ForceTarget()
    char = GetChar()
    hrp = GetHRP(char)
    alive = workspace:FindFirstChild("Alive")
    
    curveDirectionVector = CalcCurveDir(ball, target)
    finalCF = CFrame.lookAt(cam.CFrame.Position, cam.CFrame.Position + curveDirectionVector)

    snap = BuildSnap()
    
    gamePositionsSnapshot = {}
    if cam and alive then
        for _, character in ipairs(alive:GetChildren()) do
            if character:IsA("Model") and character.PrimaryPart then
                screenPos, onScreen = cam:WorldToScreenPoint(character.PrimaryPart.Position)
                gamePositionsSnapshot[character.Name] = screenPos
            end
        end
    end

    local packetSent = false

    if sendParry and not packetSent then
        local success, err = pcall(function()
            sendParry(finalCF, gamePositionsSnapshot, {cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2})
        end)
        if success then
            packetSent = true
        else
            print("Illusion Network Error: sendParry path failed execution due to " .. tostring(err))
        end
    end

    if _reverted and next(_reverted) ~= nil and not packetSent then
        local successCount = 0
        for remote, args in pairs(_reverted) do
            if type(args) ~= "table" then continue end
            local success, err = pcall(function()
                pa = {}
                pa[1] = args[1]
                pa[2] = args[2]
                pa[3] = _tokenize and _tokenize(args[2]) or args[3]
                pa[4] = isSpam and 0 or 0.5
                pa[5] = finalCF
                pa[6] = gamePositionsSnapshot
                pa[7] = {cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2}
                pa[8] = false
                if remote:IsA("RemoteEvent") then
                    remote:FireServer(unpack(pa))
                else
                    remote:InvokeServer(unpack(pa))
                end
            end)
            if success then
                successCount = successCount + 1
            else
                print("Illusion Network Error: _reverted remote path failed execution due to " .. tostring(err))
            end
        end
        if successCount > 0 then
            packetSent = true
        end
    end

    if _firstSignal and not packetSent then
        local success, err = pcall(function()
            _firstSignal:Fire(0.5, finalCF, gamePositionsSnapshot, {cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2}, false)
        end)
        if success then
            packetSent = true
        else
            print("Illusion Network Error: _firstSignal path failed execution due to " .. tostring(err))
        end
    end

    if not packetSent then
        pcall(function()
            if mouse1press then mouse1press(); task.wait(0.03); mouse1release() end
            if mouse1click then mouse1click() end
        end)
    end
end

-- NORMAL MODE — FLYTE FORMULA
--=================================================================
local function ParryNormal(ball, distance, speed, velocity, ping_ms)
    local endPosition   = 101 - math.clamp(Cfg.Accuracy, 1, 100)
    local maximumValue  = math.clamp((endPosition - 1) / 99, 0, 1)
    local offsetZ       = maximumValue * maximumValue * (3 - 2 * maximumValue) -- smoothstep
    local openError     = math.clamp(ping_ms / 10, 10, 16.7)
    local step          = math.clamp(ping_ms / 1000 * 0.5 - 0.05, 0, 0.04)
    local clampedY      = math.clamp(Cfg.ParryRange or 0, 0, 10)

    local newDenom  = 2.4 + math.clamp(speed - 9.5, 0, 990.5) * 0.002
    local center    = speed / newDenom * (0.75 + maximumValue * 0.43)
    local isInside  = center <= 9.5
    local clampedX  = isInside
        and math.min(openError, speed * (ping_ms / 1000))
        or  openError + math.min(speed * step, 30)
    local zone = clampedX + math.max(center, 9.5) + clampedY

    -- expanded zone for high accuracy
    if endPosition > 70 then
        local maxLen      = math.clamp((endPosition - 44) / 30, 0, 1)
        local index35     = maxLen * maxLen * (3 - 2 * maxLen)
        local index37     = (isInside and math.min(openError, speed * (ping_ms/1000)) or openError) + math.max(speed / newDenom * (0.75 + maximumValue * 0.2), 9.5)
        local index38     = openError + speed / newDenom * (1.4 + maximumValue * 0.2)
        local clampedDist = math.clamp((speed - 1000) / 800, 0, 1)
        local index39     = index38 + math.max(index37 - index38, 0) * clampedDist
        zone = zone + (index39 + (index37 - index39) * 0 - zone) * index35
    end

    return distance <= zone
end

--=================================================================
-- PRO MODE — BYTE FORMULA
--=================================================================
local function ParryPro(distance, speed, ping_ms)
    local acc = math.clamp(Cfg.Accuracy, 1, 100)
    local SDM
    if acc <= 50 then
        SDM = 0.8 + (acc - 1) * 0.024747474747474751
    else
        SDM = 0.8 + (acc - 1) * 0.0030303030303030303
    end
    local zone = math.clamp(ping_ms / 10, 10, 16) + speed / (2.3497562749127492 + math.clamp(speed, 0, 425) * 0.002) * SDM
    zone = zone + (Cfg.ParryRange or 0)
    return distance <= zone
end

local function StartParryLoop()

                        distCheck = (myPos - GetBallPos(b)).Magnitude
                        if distCheck < minLobbyDist and distCheck <= (Cfg.LobbyParryRange or 25) then
                            minLobbyDist = distCheck
                            mainTargetBall = b
                        end
                    end
                end
            else
                for _, b in ipairs(balls) do

                        mainTargetBall = b
                        break
                    end
                end
            end

            if not mainTargetBall or not mainTargetBall.Parent then return end
            ball = mainTargetBall

            target = ball:GetAttribute("target")
            isTargetingMe = (target == game:GetService("Players").LocalPlayer.Name)
            
            if parryLocked[ball] == true then 
                return 
            end

            if not isTargetingMe then
                local currentTarget = tostring(ball:GetAttribute("target") or "")
                if currentTarget ~= "" and currentTarget ~= game:GetService("Players").LocalPlayer.Name then
                    parryLocked[ball] = false
                    lockTime[ball] = 0
                end
                return

            if Cfg.CooldownProtection then
                currentRunningTime = os.clock()
                for targetedBall, lockedTimestamp in pairs(lockTime) do
                    if parryLocked[targetedBall] == true and lockedTimestamp and lockedTimestamp > 0 then
                        if (currentRunningTime - lockedTimestamp) > 0.7 then
                            parryLocked[targetedBall] = false
                            lockTime[targetedBall] = 0
                            if lastHitTick and lastHitTick[targetedBall] then
                                lastHitTick[targetedBall] = 0
                            end
                            if activeParryTasks and activeParryTasks[targetedBall] then
                                task.cancel(activeParryTasks[targetedBall])
                                activeParryTasks[targetedBall] = nil
                            end
                            print("Illusion Protection Warning: Cooldown Protection forced unlock on ball due to state freeze longer than 0.7s")
                        end
                    end
                end
            end           
            

            ballPos = GetBallPos(ball)
            vel     = GetBallVel(ball)
            speed   = vel.Magnitude
            dist    = (myPos - ballPos).Magnitude
            ping    = GetPing()
            tti     = dist / math.max(speed, 1)
            
            isSingularActiveNow = (char:GetAttribute("IS_EVENT_SINGULARITY") == true)
            if isSingularActiveNow then LastSingularityActiveTime = now end
            
            if (now - LastSingularityActiveTime) <= 0.1 then
                if dist <= 12 and speed <= 8 then
                    return
                end
            end
                
            blocked = false
            pcall(function()
                if Cfg.InfinityDet and Det_State and Det_State.Infinity_Active then blocked = true end
                if Cfg.ForcefieldDet and Det_State and Det_State.Forcefield_Active then blocked = true end
                if Cfg.DeathSlashDet and Det_State and Det_State.DeathSlash_Active then blocked = true end
                if Cfg.AntiPhantom and PhantomTargetData and not PhantomTargetData.Released then blocked = true end
            end)
            if blocked then return end

                    trigger = true
                end
            end

             if not trigger then
                if dist <= 5.5 then
                    trigger = true
                elseif Cfg.ParryMode == "Normal" then
                    trigger = ParryNormal(ball, dist, speed, vel, ping)
                elseif Cfg.ParryMode == "Pro" then
                    trigger = ParryPro(dist, speed, ping)
                elseif Cfg.ParryMode == "Advance" then
                    local cap = math.min(math.max(speed - 9.5, 0), 650)
                    local div = (2.4 + cap * 0.002) * (Cfg.Accuracy and (0.7 + (Cfg.Accuracy - 1) * (0.9 / 99)) or 1.1)
                    local zone = (math.clamp(ping / 10, 5, 17)) + math.max(speed / div, 9.5)
                    
                    if speed >= 450 then
                        local virex_speed_compensation = (speed / 500) * math.clamp(ping / 10, 4, 11)
                        zone = zone + virex_speed_compensation
                    end

                    local targetPlayer = ForceTarget()
                    local distToTarget = targetPlayer and targetPlayer.PrimaryPart and (myPos - targetPlayer.PrimaryPart.Position).Magnitude or math.huge

                        trigger = true
                    elseif dist <= zone and not curvedVirex then
                        trigger = true
                    end
                end
            end

            if trigger then
                if (now - lastParryTime) >= 0.002 then
                    local burst = speed > 1800 and 3 or 1
                    local executionSuccess = true
                    local executionError = ""

                    for _ = 1, burst do 
                        local success, err = pcall(function()
                            FireParry(ball, false)
                        end)
                        if not success then
                            executionSuccess = false
                            executionError = tostring(err)
                            break
                        end
                    end
                    
                    if executionSuccess then
                        lastParryTime = now
                        parryLocked[ball]  = true
                        lockTime[ball]     = now
                        lastHitTick[ball]  = now
                    else
                        parryLocked[ball]  = false
                        lockTime[ball]     = 0
                        print("Illusion Main Loop Error: FireParry execution failed inside trigger block due to " .. executionError)
                    end
                end
            end
        end, function(e) print("Illusion Main Loop Error: Physics bind crash caught due to " .. tostring(e)) end)
    end

    local loop = Cfg.ParryLoop
    if loop == "PreRender" then
        _parryConn = RunService.PreRender:Connect(loopFn)
    elseif loop == "Heartbeat" then
        _parryConn = RunService.Heartbeat:Connect(loopFn)
    else
        _parryConn = RunService.PreSimulation:Connect(loopFn)
    end
end

_spamTask = nil

function SpamLoop()
    while shared._IllusionRunning do
        local delayTime = tonumber(Cfg.SpamSpeed) or 0.005
        local enforcedDelay = math.clamp(delayTime, 0.001, 2.0)
        
        if not Cfg.AutoSpam then 
            task.wait(0.1) 
            continue 
        end
        
        local char = GetChar()
        local hrp = GetHRP(char)
        local hum = GetHum(char)
        if not hrp or not hum then 
            task.wait(0.05) 
            continue 
        end

        -- [แก้ไขจุดนี้]: เปลี่ยนเป็น return เพื่อสั่งระงับการสแกนลูกบอลทุกลูกในรอบเฟรมนี้ทันทีเมื่อติดสถานะบล็อก
        if char:GetAttribute("Pulsed") or (Det_State and Det_State.DeathSlash_Active) then
            if flyteBBState then flyteBBState.AutoSpamCarry = 0 end
            task.wait(0.016)
            return -- สั่งตัดฟังก์ชันรอบนี้ทิ้งทันทีอย่างปลอดภัย
        end

        local myPos = hrp.Position
        local balls = GetBalls()
        local now = os.clock()
        local ping = GetPing()
        local fired = false

        flyteBBState = flyteBBState or {}
        flyteBBState.AutoSpamTime = flyteBBState.AutoSpamTime or now
        local frameDelta = math.min(now - flyteBBState.AutoSpamTime, 0.066666666666666666)
        flyteBBState.AutoSpamTime = now

        local openError = math.clamp(ping / 10, 10, 16.7)
        local step = math.clamp((ping / 1000) * 0.5 - 0.05, 0, 0.04)
        local steering = ping
        local clampedY = math.clamp(tonumber(Cfg.ParryRange or 0), 0, 10)
        local maxSpamPackets = 7 

        for _, ball in ipairs(balls) do
            if not ball or not ball.Parent then continue end
            
            local ballPos = GetBallPos(ball)
            local vel = GetBallVel(ball)
            local speed = vel.Magnitude
            local ballDist = (myPos - ballPos).Magnitude
            local vertexCount = tostring(ball:GetAttribute("target") or "")

            if speed < 0.001 or ballDist < 0.1 then continue end

            local callbackFunction = (vertexCount == LP.Name) and ball:GetAttribute("from") or vertexCount
            local enemyChar = type(callbackFunction) == "string" and callbackFunction ~= "" and workspace:FindFirstChild("Alive") and workspace.Alive:FindFirstChild(callbackFunction)
            local enemyHRP = enemyChar and (enemyChar.PrimaryPart or enemyChar:FindFirstChild("HumanoidRootPart"))
            
            local cell = enemyHRP and (myPos - enemyHRP.Position).Magnitude or math.huge

            local targetStep = math.clamp(10 + steering / 10, 10, 35)
            local mathMax = #balls >= 2 and 3 or 0
            local leftChild = (Cfg.ParryType ~= "Remote") and 15 or 0

            local offsetZ = targetStep + math.min(speed / 6, 95) + mathMax + leftChild
            local newDenominator = targetStep + math.min(speed / 6, 30) + mathMax + leftChild
            
            local isPinned = cell <= newDenominator and ballDist <= offsetZ

            if isPinned and not Cfg.OldSpam then
                local maximumValue = 1
                if Cfg.DynamicSpam then
                    maximumValue = math.clamp(math.ceil(math.min(math.max(cell / newDenominator, ballDist / offsetZ), 2 * math.max(cell, ballDist) / speed * math.min(0.5 + steering / 1000, 0.75)) * 3), 1, 3)
                else
                    maximumValue = math.clamp(math.round(Cfg.SpamThreshold or 2), 1, 4)
                end

                local offsetZ_Time = now - math.min(0.5 + steering / 1000, 0.75)
                while SpamParries[1] and SpamParries[1] < offsetZ_Time do
                    table.remove(SpamParries, 1)
                end

                if #SpamParries < maximumValue then
                    isPinned = false
                end
            end

            if isPinned or Cfg.OldSpam then
                xpcall(function()
                    if Cfg.ParryType == "Remote" and sendParry then
                        flyteBBState.AutoSpamCarry = (flyteBBState.AutoSpamCarry or 0) + math.max(frameDelta * 120 - 1, 0)
                        local packetCount = math.min(math.floor(flyteBBState.AutoSpamCarry), maxSpamPackets)
                        flyteBBState.AutoSpamCarry = math.min(flyteBBState.AutoSpamCarry - packetCount, maxSpamPackets)
                        
                        for i = 1, (packetCount + 1) do
                            FireParry(ball, true)
                        end
                    else
                        flyteBBState.AutoSpamCarry = 0
                        FireParry(ball, true)
                    end
                    
                    SpamParryTime = now
                    if not Cfg.OldSpam then
                        table.insert(SpamParries, now)
                    end
                    fired = true
                end, function(err)
                    print("Illusion Engine Error: Enhanced Auto Spam execution failed due to " .. tostring(err))
                end)
                
                if fired then break end
            end
        end

        task.wait(fired and enforcedDelay or 0.011) 
    end
end

if _spamTask then task.cancel(_spamTask) end

local _flyConn, _flyBV, _flyGyro = nil, nil, nil

local function StartFly()
    local char = GetChar()
    local hrp = GetHRP(char)
    local hum = GetHum(char)
    if not hrp or not hum then return end
    
    hum.PlatformStand = true
    
    if _flyBV then _flyBV:Destroy() end
    if _flyGyro then _flyGyro:Destroy() end
    if _flyConn then _flyConn:Disconnect() end
    
    _flyBV = Instance.new("BodyVelocity")
    _flyBV.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    _flyBV.Velocity = Vector3.zero
    _flyBV.Parent = hrp
    
    _flyGyro = Instance.new("BodyGyro")
    _flyGyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    _flyGyro.P = 9e4
    _flyGyro.Parent = hrp
    
    _flyConn = RunService.RenderStepped:Connect(function()
        if not Cfg.FlyEnabled or not hrp or not hum or not _flyBV or not _flyGyro then return end
        
        hum.PlatformStand = true
        local cam = workspace.CurrentCamera
        if not cam then return end
        
        _flyGyro.CFrame = cam.CFrame
        
        local moveVector = Vector3.zero
        local controlModule = nil
        
        pcall(function()
            controlModule = require(LP.PlayerScripts:FindFirstChild("PlayerModule"):FindFirstChild("ControlModule"))
        end)
        
        if controlModule and controlModule.GetMoveVector then
            local rawVector = controlModule:GetMoveVector()
            moveVector = Vector3.new(rawVector.X, 0, rawVector.Z)
        else
            moveVector = hum.MoveDirection
        end
        
        if moveVector.Magnitude == 0 then
            if UIS:IsKeyDown(Enum.KeyCode.W) then moveVector = moveVector + Vector3.new(0, 0, -1) end
            if UIS:IsKeyDown(Enum.KeyCode.S) then moveVector = moveVector + Vector3.new(0, 0, 1) end
            if UIS:IsKeyDown(Enum.KeyCode.A) then moveVector = moveVector + Vector3.new(-1, 0, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.D) then moveVector = moveVector + Vector3.new(1, 0, 0) end
        end
        
        local sp = Cfg.FlySpeed or 50
        local direction = cam.CFrame:VectorToWorldSpace(moveVector)
        local verticalVel = 0
        
        if UIS:IsKeyDown(Enum.KeyCode.Space) then verticalVel = sp end
        if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then verticalVel = -sp end
        
        _flyBV.Velocity = (direction * sp) + Vector3.new(0, verticalVel, 0)
    end)
end

    if _flyBV then _flyBV:Destroy(); _flyBV = nil end
    if _flyGyro then _flyGyro:Destroy(); _flyGyro = nil end
    
    local hum = GetHum(GetChar())
    if hum then 
        hum.PlatformStand = false 
    end
end

local RenderCache = {}
local NoRenderConn = nil

local FX_Blacklist = {
    BallEmit = true, InfinityFX = true, Tornado = true, Barrier = true,
    SpiritDeflect = true, dash = true, Wave = true
}

-- ฟังชันก์ซ่อนวัตถุและบันทึกสถานะเดิมเก็บไว้ใน Cache
local function ApplyNoRenderFX(obj)
    if not obj or not obj.Parent then return end
    if FX_Blacklist[obj.Name] then return end -- ข้ามวัตถุที่อยู่ใน Blacklist
    
    local className = obj.ClassName
    
    if className == "ParticleEmitter" or className == "Beam" or className == "Trail" or 
       className == "Smoke" or className == "Fire" or className == "Sparkles" then
        
        if RenderCache[obj] == nil then
            RenderCache[obj] = { Enabled = obj.Enabled }
        end
        obj.Enabled = false
        if className == "ParticleEmitter" then pcall(function() obj:Clear() end) end
        
    elseif className == "Decal" or className == "Texture" then
        if RenderCache[obj] == nil then
            RenderCache[obj] = { Transparency = obj.Transparency }
        end
        obj.Transparency = 1
        
    elseif className == "SurfaceGui" or className == "BillboardGui" then
        if obj.Name == "Ability_ESP" or obj.Name == "Illusion_Plr_ESP" or obj.Name == "ComboCounter" or obj.Name == "FuryCharge" then
            return
        end
        
        local aliveFolder = workspace:FindFirstChild("Alive")
        if aliveFolder and obj:IsDescendantOf(aliveFolder) then
            return
        end

        if obj:IsDescendantOf(workspace) and not obj:IsDescendantOf(workspace:FindFirstChild("Alive") or game) then
            if RenderCache[obj] == nil then
                RenderCache[obj] = { Enabled = obj.Enabled }
            end
            obj.Enabled = false
        end
    elseif obj:IsA("BasePart") then
        if obj:IsDescendantOf(workspace:FindFirstChild("Alive") or game) or obj:IsDescendantOf(workspace:FindFirstChild("Balls") or game) or obj:IsDescendantOf(workspace:FindFirstChild("TrainingBalls") or game) then
            return
        end
        
        local lowerName = string.lower(obj.Name)
        if lowerName == "sord" or lowerName == "blade" or lowerName:find("handle") then return end
        
        if RenderCache[obj] == nil then
            RenderCache[obj] = {
                LocalTransparencyModifier = obj.LocalTransparencyModifier,
                CanCollide = obj.CanCollide,
                CanTouch = obj.CanTouch,
                CanQuery = obj.CanQuery
            }
        end
        
        pcall(function()
            obj.LocalTransparencyModifier = 1
            obj.CanCollide = false
            obj.CanTouch = false
            obj.CanQuery = false
        end)
    end
end

local function RestoreRenderFX()
    if not RenderCache then return end
    for obj, originalProperties in pairs(RenderCache) do
        if obj and obj.Parent then
            pcall(function()
                for property, val in pairs(originalProperties) do
                    obj[property] = val
                end
            end)
        end
    end
    table.clear(RenderCache)
end

local function SetNoRender(on)
    if NoRenderConn then NoRenderConn:Disconnect(); NoRenderConn = nil end
    
    if on then
        for _, obj in ipairs(workspace:GetDescendants()) do

        end
        
        NoRenderConn = workspace.DescendantAdded:Connect(function(obj)
            if not Cfg.NoRender then return end
            pcall(ApplyNoRenderFX, obj)
        end)
    else
        pcall(RestoreRenderFX)
    end
end

local LastLobbyParryTime = 0
local LobbyParryState = {} -- เก็บข้อมูลสถานะของลูกบอลแต่ละลูกแบบเจาะลึก

local function ProcessLobbyParry()
    xpcall(function()
        if not shared._IllusionRunning or not Cfg.AutoLobbyParry then return end
        
        local char = GetChar()
        local hrp = GetHRP(char)
        local hum = GetHum(char)
        if not hrp or not hum or hum.Health <= 0 then return end
        
        if Cfg.AutoParry and ball and ball.Parent and ball:GetAttribute("target") == game:GetService("Players").LocalPlayer.Name then
            return 
        end

        local myPos = hrp.Position
        local lobbyBalls = GetLobbyTrainingBalls()
        if #lobbyBalls == 0 then return end
        
        local now   = os.clock()
        local ping  = GetPing()
        
        -- คูลดาวน์พื้นฐานขั้นต่ำเพื่อป้องกันการเบิ้ลระดับฮาร์ดแวร์
        local hardCooldown = math.clamp((ping / 1000) + 0.12, 0.2, 0.5)

        for _, b in ipairs(lobbyBalls) do
            if b and b.Parent then
                local ballPos = GetBallPos(b)
                local vel     = GetBallVel(b)
                local speed   = vel.Magnitude
                local dist    = (myPos - ballPos).Magnitude
                
                -- สร้างโปรไฟล์จัดเก็บสถานะลูกบอลถ้ายังไม่มี
                if not LobbyParryState[b] then
                    LobbyParryState[b] = {
                        LastFiredTime = 0,
                        LastDirection = vel.Unit,
                        IsProcessed = false
                    }
                end
                
                local bState = LobbyParryState[b]
                
                -- ระบบล้างตัวล็อก: ถ้าระยะห่างลูกบอลพุ่งไกลออกไป หรือทิศทางความเร็วเปลี่ยนไปในทางตรงกันข้าม (เราตีสวนไปแล้ว)
                -- ให้ถือว่าลูกบอลจังหวะนั้นจบลงแล้ว ให้ปลดล็อกได้
                local isMovingAway = vel.Unit:Dot(bState.LastDirection) < 0 or (ballPos - myPos).Magnitude > dist + 1
                if isMovingAway and bState.IsProcessed then
                    bState.IsProcessed = false
                end

                if dist <= (Cfg.LobbyParryRange or 25) then
                    
                    -- บล็อกการเบิ้ลคลิกซ้ำ: 
                    -- 1. ล็อกไว้ถ้าเวลากดครั้งล่าสุดยังไม่พ้น Hard Cooldown
                    -- 2. ล็อกไว้ถ้าสเตต IsProcessed ยังค้างอยู่ (แปลว่าลูกบอลยังไม่ได้ถูกตีสวนกลับไป ทิศทางยังเหมือนเดิม)
                    if (now - bState.LastFiredTime) < hardCooldown or bState.IsProcessed then
                        continue 
                    end
                    
                    -- ระบบกันสเตตค้าง (Safety Reset กรณีเซิร์ฟเวอร์เคลียร์ตำแหน่งไม่ทันจริงๆ)
                    if (now - bState.LastFiredTime) > 1.2 then
                        bState.IsProcessed = false
                    end

                    -- ปล่อยคำสั่งพาร์รี่
                    if (now - LastLobbyParryTime) >= 0.005 then
                        bState.IsProcessed = true
                        bState.LastFiredTime = now
                        bState.LastDirection = vel.Unit -- บันทึกทิศทางลูกบอลตอนที่เราตีออกไป
                        LastLobbyParryTime = now
                        
                        xpcall(function()
                            FireParry(b, false)
                        end, function(err)
                            bState.IsProcessed = false -- ปลดล็อกหากยิงแพ็กเก็ตไม่ผ่าน
                            print("Illusion Network Error: Lobby FireParry failed execution due to " .. tostring(err))
                        end)
                        break
                    end
                else
                    -- ล้างข้อมูลชั่วคราวเมื่อลูกบอลหลุดระยะ
                    if not bState.IsProcessed then
                        bState.LastFiredTime = 0
                    end
                end
            end
        end
    end, function(e)
        print("Illusion Lobby Loop Error: Connection handling process crashed due to " .. tostring(e))
    end)
end

local TargetMorphUser = ""

local function ApplyAvatarMorph(targetInput)
    if not targetInput or targetInput == "" then return end
    local character = game:GetService("Players").LocalPlayer.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local head = character:FindFirstChild("Head")
    if not humanoid or not head then return end
    
    local num = tonumber(targetInput)
    if not num then
        pcall(function()
            num = game:GetService("Players"):GetUserIdFromNameAsync(targetInput)
        end)
    end
    if not num then return end

    local ok, result = pcall(function()
        return game:GetService("Players"):CreateHumanoidModelFromUserId(num)
    end)

    if ok and result then
        local head2 = result:FindFirstChild("Head")
        local humanoid2 = result:FindFirstChildOfClass("Humanoid")

        if head2 then
            local decal = head:FindFirstChildOfClass("Decal")
            local decal2 = head2:FindFirstChildOfClass("Decal")
            if decal2 then
                if not decal then
                    decal = Instance.new("Decal")
                    decal.Name = "face"
                    decal.Parent = head
                end
                decal.Texture = decal2.Texture
            elseif decal then
                decal:Destroy()
            end

            local specialMesh = head:FindFirstChildOfClass("SpecialMesh")
            local specialMesh2 = head2:FindFirstChildOfClass("SpecialMesh")
            if specialMesh2 then
                if not specialMesh then
                    specialMesh = Instance.new("SpecialMesh")
                    specialMesh.Parent = head
                end
                specialMesh.MeshId = specialMesh2.MeshId
                specialMesh.TextureId = specialMesh2.TextureId
                specialMesh.Scale = specialMesh2.Scale
            end
        end

        for _, child in ipairs(character:GetChildren()) do
            if child:IsA("Accessory") or child:IsA("Clothing") or child:IsA("ShirtGraphic") or child:IsA("BodyColors") or child:IsA("CharacterMesh") then
                child:Destroy()
            end
        end

        local bodyColors = result:FindFirstChildOfClass("BodyColors")
        if bodyColors then bodyColors:Clone().Parent = character end

        if humanoid.RigType == Enum.HumanoidRigType.R15 and humanoid2 then
            for _, v90 in ipairs({ "BodyHeightScale", "BodyWidthScale", "BodyDepthScale", "HeadScale", "ProportionScale" }) do
                local v91 = humanoid2:FindFirstChild(v90)
                local v92 = humanoid:FindFirstChild(v90)
                if v91 and v92 then v92.Value = v91.Value end
            end
        end

        if humanoid.RigType == Enum.HumanoidRigType.R6 then
            for _, child in ipairs(result:GetChildren()) do
                if child:IsA("CharacterMesh") then child:Clone().Parent = character end
            end
        else
            for _, child in ipairs(result:GetChildren()) do
                if child:IsA("MeshPart") then
                    local v90 = character:FindFirstChild(child.Name)
                    if v90 and v90:IsA("MeshPart") then
                        v90.MeshId = child.MeshId
                        v90.TextureID = child.TextureID
                    end
                end
            end
        end

        for _, child in ipairs(result:GetChildren()) do
            if child:IsA("Accessory") then
                local clone = child:Clone()
                clone.Parent = character
                local handle = clone:FindFirstChild("Handle")
                if handle and handle:IsA("BasePart") then
                    handle.CanCollide = false
                    if not handle:FindFirstChildOfClass("WrapLayer") then
                        local attachment = handle:FindFirstChildOfClass("Attachment")
                        local v90 = nil
                        if attachment then
                            for _, child2 in ipairs(character:GetChildren()) do
                                if child2:IsA("BasePart") and child2:FindFirstChild(attachment.Name) then
                                    v90 = child2:FindFirstChild(attachment.Name)
                                    break
                                end
                            end
                        end

                        if v90 then
                            handle.CFrame = v90.WorldCFrame * attachment.CFrame:Inverse()
                            local weldConstraint = Instance.new("WeldConstraint")
                            weldConstraint.Part0 = handle
                            weldConstraint.Part1 = v90.Parent
                            weldConstraint.Parent = handle
                        else
                            local torso = head or character:FindFirstChild("Torso") or character:FindFirstChild("UpperTorso")
                            if torso then
                                handle.CFrame = torso.CFrame
                                local weldConstraint = Instance.new("WeldConstraint")
                                weldConstraint.Part0 = handle
                                weldConstraint.Part1 = torso
                                weldConstraint.Parent = handle
                            end
                        end
                    end
                end
            elseif child:IsA("Clothing") or child:IsA("ShirtGraphic") then
                child:Clone().Parent = character
            end
        end

        result:Destroy()
    end
end

            p.CharacterAdded:Connect(function()
                if Cfg.ESPAbility then task.spawn(ApplyPlayerESP, p) end

    local cfgKey = label:gsub("%s+", "")

    local val = default or false
    if Cfg[cfgKey] ~= nil then
        val = Cfg[cfgKey]
    end
    
    local Row = Instance.new("Frame")
    Row.Size = UDim2.new(1,0,0,desc and 44 or 32)
    Row.BackgroundColor3 = CC
    Row.Parent = Pages[page]
    Rnd(Row,6)

    L.Size = UDim2.new(1,-65,0,16)

        local D = Lbl(Row,desc,9,CS)
        D.Position = UDim2.fromOffset(10,22)
        D.Size = UDim2.new(1,-65,0,14)
    end

    local Tr = Instance.new("Frame")
    Tr.Size = UDim2.fromOffset(32,16)
    Tr.Position = UDim2.new(1,-42,0.5,-8)
    Tr.BackgroundColor3 = val and CG or Color3.fromRGB(35,35,52)
    Tr.Parent = Row
    Rnd(Tr,20)

    local Kn = Instance.new("Frame")
    Kn.Size = UDim2.fromOffset(10,10)
    Kn.Position = val and UDim2.new(1,-13,0.5,-5) or UDim2.fromOffset(3,3)
    Kn.BackgroundColor3 = Color3.new(1,1,1)
    Kn.Parent = Tr
    Rnd(Kn,20)

    local TB = Instance.new("TextButton")
    TB.Size = UDim2.fromScale(1,1)
    TB.BackgroundTransparency = 1

    TB.Parent = Tr

    TB.MouseButton1Click:Connect(function()

            if cb then cb(val) end
        end, function(err)

        end)

function Slider(page,label,min,max,default,fmt,cb)
    local cfgKey = label:gsub("%s+", "")    
    if label == "Cooldown Value ×100" then cfgKey = "CooldownValue" end
    
    local val = default
    if Cfg[cfgKey] ~= nil then
        if label == "Cooldown Value ×100" then
            val = Cfg[cfgKey] * 100
        else
            val = Cfg[cfgKey]
        end
    end
    
    local H = Instance.new("Frame")
    H.Size = UDim2.new(1,0,0,58)
    H.BackgroundColor3 = CC
    H.Parent = Pages[page]
    Rnd(H,8)
    Str(H,CST,1)
    
    local L = Lbl(H,label,12,CW)
    L.Position = UDim2.fromOffset(12,6)
    L.Size = UDim2.new(1,-70,0,16)
    
    local VL = Lbl(H,(fmt or "%d"):format(val),11,CA,Enum.TextXAlignment.Right)
    VL.Position = UDim2.new(1,-58,0,6)
    VL.Size = UDim2.fromOffset(48,16)
    
    local TBG = Instance.new("Frame")
    TBG.Size = UDim2.new(1,-24,0,5)
    TBG.Position = UDim2.fromOffset(12,35)
    TBG.BackgroundColor3 = Color3.fromRGB(35,35,52)
    TBG.Parent = H
    Rnd(TBG,4)
    
    local Fill = Instance.new("Frame")
    Fill.BackgroundColor3 = CA
    Fill.Parent = TBG
    Rnd(Fill,4)
    
    local Knob = Instance.new("Frame")
    Knob.Size = UDim2.fromOffset(11,11)
    Knob.AnchorPoint = Vector2.new(0.5,0.5)
    Knob.BackgroundColor3 = CW
    Knob.Parent = TBG
    Rnd(Knob,20)
    
    local function UpdateVisuals(currentVal)
        local p = math.clamp((currentVal - min) / (max - min), 0, 1)
        Tw(Fill, 0.1, {Size = UDim2.new(p, 0, 1, 0)})
        Tw(Knob, 0.1, {Position = UDim2.new(p, 0, 0.5, 0)})
        VL.Text = (fmt or "%d"):format(currentVal)
    end
    
    UpdateVisuals(val)
    
    local drg = false
    
    local function Upd(px)
        xpcall(function()
            local p = math.clamp((px - TBG.AbsolutePosition.X) / TBG.AbsoluteSize.X, 0, 1)
            val = math.floor(min + (max - min) * p)
            
            if label == "Cooldown Value ×100" then
                Cfg[cfgKey] = val / 100
            else
                Cfg[cfgKey] = val
            end
            
            UpdateVisuals(val)
            
            if cb then cb(val) end
        end, function()
            print("Error updating visual elements positions inside slider interface component")
        end)
    end
    
    TBG.InputBegan:Connect(function(i) 
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then 
            drg = true 
            Upd(i.Position.X) 
        end 
    end)
    
    UIS.InputChanged:Connect(function(i) 
        if drg and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then 
            Upd(i.Position.X) 
        end 
    end)
    
    UIS.InputEnded:Connect(function(i) 
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then 
            if drg then
                drg = false 
                SaveConfig()
            end
        end 
    end)
    
    return {
        Get = function() return val end,
        Set = function(_, v)
            if label == "Cooldown Value ×100" then
                val = v * 100
                Cfg[cfgKey] = v
            else
                val = v
                Cfg[cfgKey] = v
            end
            UpdateVisuals(val)
            if cb then cb(val) end
            SaveConfig()
        end
    }
end

            d = true
            moved = false
            ds = i.Position
            sp = btn.Position
        end
    end)
    
    UIS.InputChanged:Connect(function(i)
        if not d then return end
        if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
            local dl = i.Position - ds
            if dl.Magnitude > 6 then 
                moved = true 
            end
            if moved then
                local vp = workspace.CurrentCamera.ViewportSize
                btn.Position = UDim2.new(
                    sp.X.Scale,
                    math.clamp(sp.X.Offset + dl.X, 0, vp.X - btn.AbsoluteSize.X),
                    sp.Y.Scale,
                    math.clamp(sp.Y.Offset + dl.Y, 0, vp.Y - btn.AbsoluteSize.Y)
                )
            end
        end
    end)
    
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            if d then
                d = false
                if not moved then 
                    xpcall(function()
                        onTap()
                    end, function(err)
                        print("Float button tap action failed: " .. tostring(err))
                    end)
                end
            end
        end
    end)
end
